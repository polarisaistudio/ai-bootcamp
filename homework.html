# model_comparison_homework.py
"""
作业1：模型对比报告
业务场景：智能客服对话系统
对比模型：GPT-3.5-turbo, GPT-4o-mini, GPT-4o
"""

import openai
import time
import pandas as pd
from datetime import datetime
from typing import List, Dict
import json

class ModelComparisonReport:
    """模型对比测试与报告生成系统"""

    def __init__(self, api_key: str):
        self.client = openai.OpenAI(api_key=api_key)
        self.results = []

    def test_model(self,
                   model: str,
                   prompt: str,
                   temperature: float = 0.7) -> Dict:
        """测试单个模型的响应"""

        start_time = time.time()

        try:
            response = self.client.chat.completions.create(
                model=model,
                messages=[
                    {"role": "system", "content": "你是一个专业的客服代表"},
                    {"role": "user", "content": prompt}
                ],
                temperature=temperature,
                max_tokens=200
            )

            elapsed_time = time.time() - start_time

            return {
                'success': True,
                'response': response.choices[0].message.content,
                'tokens': response.usage.total_tokens,
                'prompt_tokens': response.usage.prompt_tokens,
                'completion_tokens': response.usage.completion_tokens,
                'time': elapsed_time,
                'cost': self._calculate_cost(model, response.usage.total_tokens)
            }

        except Exception as e:
            return {
                'success': False,
                'error': str(e),
                'tokens': 0,
                'time': 0,
                'cost': 0
            }

    def _calculate_cost(self, model: str, tokens: int) -> float:
        """计算API调用成本"""
        pricing = {
            'gpt-3.5-turbo': 0.001,      # $0.001 per 1K tokens
            'gpt-4o-mini': 0.0006,       # $0.0006 per 1K tokens
            'gpt-4o': 0.01,              # $0.01 per 1K tokens
        }
        price_per_1k = pricing.get(model, 0.001)
        return (tokens / 1000) * price_per_1k

    def run_comparison(self,
                      models: List[str],
                      test_prompts: List[str],
                      temperature: float = 0.7):
        """运行完整对比测试"""

        print("🚀 开始模型对比测试")
        print("=" * 70)

        for model in models:
            print(f"\n📊 测试模型: {model}")
            print("-" * 70)

            for idx, prompt in enumerate(test_prompts, 1):
                print(f"\n  测试 {idx}/{len(test_prompts)}: {prompt[:40]}...")

                result = self.test_model(model, prompt, temperature)

                if result['success']:
                    print(f"    ✓ 成功")
                    print(f"    ⏱️  响应时间: {result['time']:.2f}s")
                    print(f"    🔤 Token数: {result['tokens']}")
                    print(f"    💰 成本: ${result['cost']:.6f}")

                    # 评估响应质量（1-10分）
                    quality_score = self._evaluate_quality(
                        prompt,
                        result['response']
                    )
                    print(f"    ⭐ 质量评分: {quality_score}/10")

                    # 保存结果
                    self.results.append({
                        'model': model,
                        'prompt': prompt,
                        'response': result['response'],
                        'quality_score': quality_score,
                        'response_time': result['time'],
                        'total_tokens': result['tokens'],
                        'prompt_tokens': result['prompt_tokens'],
                        'completion_tokens': result['completion_tokens'],
                        'cost': result['cost'],
                        'timestamp': datetime.now().isoformat()
                    })
                else:
                    print(f"    ✗ 失败: {result['error']}")

                # 避免速率限制
                time.sleep(1)

    def _evaluate_quality(self, prompt: str, response: str) -> float:
        """评估响应质量（简化版）"""
        score = 10.0

        # 检查响应长度
        if len(response) < 20:
            score -= 3
        elif len(response) > 500:
            score -= 1

        # 检查是否包含礼貌用语
        polite_words = ['您', '请', '谢谢', '感谢', '抱歉']
        if not any(word in response for word in polite_words):
            score -= 1

        # 检查是否有结构化内容
        if '：' in response or '。' in response:
            score += 0.5

        return min(max(score, 0), 10)

    def generate_report(self) -> str:
        """生成Markdown格式的对比报告"""

        if not self.results:
            return "# 错误：没有测试结果\n"

        df = pd.DataFrame(self.results)

        report = []
        report.append("# 模型对比测试报告\n")
        report.append(f"**测试时间**: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
        report.append(f"**业务场景**: 智能客服对话系统\n")
        report.append(f"**测试模型数**: {df['model'].nunique()}\n")
        report.append(f"**测试样本数**: {len(df)}\n\n")

        # 1. 总体对比表
        report.append("## 1. 模型总体表现对比\n\n")

        summary = df.groupby('model').agg({
            'quality_score': 'mean',
            'response_time': 'mean',
            'total_tokens': 'mean',
            'cost': 'sum'
        }).round(4)

        report.append("| 模型 | 平均质量 | 平均响应时间(s) | 平均Token数 | 总成本($) |")
        report.append("|------|----------|-----------------|-------------|-----------|")

        for model, row in summary.iterrows():
            report.append(
                f"| {model} | {row['quality_score']:.2f} | "
                f"{row['response_time']:.2f} | {row['total_tokens']:.0f} | "
                f"${row['cost']:.6f} |"
            )

        report.append("\n")

        # 2. 详细测试结果
        report.append("## 2. 详细测试结果\n\n")

        for model in df['model'].unique():
            report.append(f"### {model}\n\n")

            model_df = df[df['model'] == model]

            for idx, row in model_df.iterrows():
                report.append(f"**测试 {idx + 1}**\n")
                report.append(f"- **Prompt**: {row['prompt']}\n")
                report.append(f"- **Response**: {row['response'][:100]}...\n")
                report.append(f"- **质量评分**: {row['quality_score']}/10\n")
                report.append(f"- **响应时间**: {row['response_time']:.2f}s\n")
                report.append(f"- **Token使用**: {row['total_tokens']} "
                            f"(输入: {row['prompt_tokens']}, "
                            f"输出: {row['completion_tokens']})\n")
                report.append(f"- **成本**: ${row['cost']:.6f}\n\n")

        # 3. 性能分析
        report.append("## 3. 性能分析\n\n")

        # 找出最佳模型
        best_quality = summary['quality_score'].idxmax()
        best_speed = summary['response_time'].idxmin()
        best_cost = summary['cost'].idxmin()

        report.append(f"- **质量最佳**: {best_quality} "
                     f"(平均评分: {summary.loc[best_quality, 'quality_score']:.2f})\n")
        report.append(f"- **速度最快**: {best_speed} "
                     f"(平均响应: {summary.loc[best_speed, 'response_time']:.2f}s)\n")
        report.append(f"- **成本最低**: {best_cost} "
                     f"(总成本: ${summary.loc[best_cost, 'cost']:.6f})\n\n")

        # 4. 结论与建议
        report.append("## 4. 结论与建议\n\n")
        report.append("### 综合分析\n\n")

        # 计算性价比 (质量/成本)
        summary['cost_efficiency'] = summary['quality_score'] / (summary['cost'] * 1000)
        best_efficiency = summary['cost_efficiency'].idxmax()

        report.append(f"基于本次测试，**{best_efficiency}** "
                     f"在客服场景下表现出最佳的性价比。\n\n")

        report.append("### 使用建议\n\n")
        report.append("1. **高频简单查询**: 使用 GPT-3.5-turbo（成本低，响应快）\n")
        report.append("2. **复杂问题处理**: 使用 GPT-4o（质量高，理解能力强）\n")
        report.append("3. **平衡性能与成本**: 使用 GPT-4o-mini（综合性价比最优）\n")
        report.append("4. **成本优化策略**: 结合FAQ系统 + 智能路由可节省90%以上成本\n\n")

        # 5. 附录
        report.append("## 5. 附录\n\n")
        report.append("### 测试环境\n")
        report.append("- Python版本: 3.11+\n")
        report.append("- OpenAI库版本: 1.12.0\n")
        report.append("- Temperature: 0.7\n")
        report.append("- Max Tokens: 200\n\n")

        return '\n'.join(report)

    def save_report(self, filename: str = None):
        """保存报告到文件"""
        if filename is None:
            filename = f"model_comparison_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.md"

        report_content = self.generate_report()

        with open(filename, 'w', encoding='utf-8') as f:
            f.write(report_content)

        print(f"\n✅ 报告已保存: {filename}")

        # 同时保存原始数据为CSV
        csv_filename = filename.replace('.md', '.csv')
        df = pd.DataFrame(self.results)
        df.to_csv(csv_filename, index=False, encoding='utf-8-sig')

        print(f"✅ 原始数据已保存: {csv_filename}")

# token_optimization_homework.py
"""
作业2：Token优化实践
目标：将原始prompt从200+ tokens优化到<100 tokens，保持语义不变
"""

import openai
import tiktoken
from typing import Dict, List
import pandas as pd

class TokenOptimizer:
    """Token优化工具类"""

    def __init__(self, api_key: str):
        self.client = openai.OpenAI(api_key=api_key)
        self.encoder = tiktoken.get_encoding("cl100k_base")

    def count_tokens(self, text: str) -> int:
        """计算文本的token数量"""
        return len(self.encoder.encode(text))

    def analyze_prompt(self, prompt: str) -> Dict:
        """分析prompt的token使用情况"""
        tokens = self.encoder.encode(prompt)
        token_count = len(tokens)

        # Token分解
        token_breakdown = []
        for token in tokens[:20]:  # 只显示前20个
            decoded = self.encoder.decode([token])
            token_breakdown.append(decoded)

        return {
            'text': prompt,
            'token_count': token_count,
            'char_count': len(prompt),
            'token_breakdown': token_breakdown,
            'efficiency': len(prompt) / token_count if token_count > 0 else 0
        }

    def test_prompt_quality(self, prompt: str, test_case: str) -> Dict:
        """测试prompt的响应质量"""
        try:
            response = self.client.chat.completions.create(
                model="gpt-3.5-turbo",
                messages=[
                    {"role": "system", "content": prompt},
                    {"role": "user", "content": test_case}
                ],
                temperature=0.7,
                max_tokens=150
            )

            return {
                'success': True,
                'response': response.choices[0].message.content,
                'tokens_used': response.usage.total_tokens,
                'cost': (response.usage.total_tokens / 1000) * 0.001
            }
        except Exception as e:
            return {
                'success': False,
                'error': str(e)
            }

    def compare_prompts(self,
                       original: str,
                       optimized: str,
                       test_cases: List[str]) -> Dict:
        """对比原始和优化后的prompt"""

        print("🔬 开始Token优化对比测试")
        print("=" * 70)

        # 分析Token使用
        print("\n📊 Token分析")
        print("-" * 70)

        original_analysis = self.analyze_prompt(original)
        optimized_analysis = self.analyze_prompt(optimized)

        print(f"\n原始Prompt:")
        print(f"  文本: {original[:60]}...")
        print(f"  字符数: {original_analysis['char_count']}")
        print(f"  Token数: {original_analysis['token_count']}")

        print(f"\n优化后Prompt:")
        print(f"  文本: {optimized[:60]}...")
        print(f"  字符数: {optimized_analysis['char_count']}")
        print(f"  Token数: {optimized_analysis['token_count']}")

        # 计算优化效果
        token_reduction = original_analysis['token_count'] - optimized_analysis['token_count']
        reduction_percent = (token_reduction / original_analysis['token_count']) * 100

        print(f"\n✨ 优化效果:")
        print(f"  Token减少: {token_reduction}")
        print(f"  减少比例: {reduction_percent:.1f}%")

        # 质量对比测试
        print("\n🎯 质量对比测试")
        print("-" * 70)

        results = {
            'original_analysis': original_analysis,
            'optimized_analysis': optimized_analysis,
            'reduction': {
                'tokens': token_reduction,
                'percent': reduction_percent
            },
            'quality_tests': []
        }

        for idx, test_case in enumerate(test_cases, 1):
            print(f"\n测试用例 {idx}: {test_case[:40]}...")

            # 测试原始prompt
            print("  原始Prompt响应...", end="")
            original_result = self.test_prompt_quality(original, test_case)
            if original_result['success']:
                print(f" ✓ ({original_result['tokens_used']} tokens)")
            else:
                print(f" ✗ 失败")

            # 测试优化prompt
            print("  优化Prompt响应...", end="")
            optimized_result = self.test_prompt_quality(optimized, test_case)
            if optimized_result['success']:
                print(f" ✓ ({optimized_result['tokens_used']} tokens)")
            else:
                print(f" ✗ 失败")

            # 对比结果
            if original_result['success'] and optimized_result['success']:
                quality_score = self._compare_quality(
                    original_result['response'],
                    optimized_result['response']
                )

                token_saving = original_result['tokens_used'] - optimized_result['tokens_used']
                cost_saving = original_result['cost'] - optimized_result['cost']

                print(f"  语义相似度: {quality_score:.1f}%")
                print(f"  Token节省: {token_saving}")
                print(f"  成本节省: ${cost_saving:.6f}")

                results['quality_tests'].append({
                    'test_case': test_case,
                    'original_response': original_result['response'],
                    'optimized_response': optimized_result['response'],
                    'quality_score': quality_score,
                    'token_saving': token_saving,
                    'cost_saving': cost_saving
                })

        return results

    def _compare_quality(self, response1: str, response2: str) -> float:
        """简单的语义相似度评估"""
        # 简化版：基于长度和关键词重叠
        words1 = set(response1.lower().split())
        words2 = set(response2.lower().split())

        overlap = len(words1 & words2)
        total = len(words1 | words2)

        similarity = (overlap / total * 100) if total > 0 else 0
        return similarity

    def generate_optimization_report(self, results: Dict) -> str:
        """生成优化报告"""
        report = []

        report.append("# Token优化实践报告\n")
        report.append("## 1. 优化前后对比\n")

        orig = results['original_analysis']
        opt = results['optimized_analysis']

        report.append("| 指标 | 原始Prompt | 优化后Prompt | 改进 |")
        report.append("|------|------------|--------------|------|")
        report.append(f"| Token数 | {orig['token_count']} | {opt['token_count']} | "
                     f"-{results['reduction']['tokens']} ({results['reduction']['percent']:.1f}%) |")
        report.append(f"| 字符数 | {orig['char_count']} | {opt['char_count']} | "
                     f"-{orig['char_count'] - opt['char_count']} |")

        report.append("\n## 2. 原始Prompt\n")
        report.append(f"```\n{orig['text']}\n```\n")

        report.append("## 3. 优化后Prompt\n")
        report.append(f"```\n{opt['text']}\n```\n")

        report.append("## 4. 优化策略\n")
        report.append("1. 删除冗余词汇和重复表述\n")
        report.append("2. 使用更简洁的句式\n")
        report.append("3. 保留核心指令和关键信息\n")
        report.append("4. 移除不必要的礼貌用语\n\n")

        report.append("## 5. 质量验证\n")

        if results['quality_tests']:
            avg_similarity = sum(t['quality_score'] for t in results['quality_tests']) / len(results['quality_tests'])
            total_token_saving = sum(t['token_saving'] for t in results['quality_tests'])

            report.append(f"- 平均语义相似度: {avg_similarity:.1f}%\n")
            report.append(f"- 总Token节省: {total_token_saving}\n")
            report.append(f"- 测试用例数: {len(results['quality_tests'])}\n\n")

        report.append("## 6. 结论\n")
        report.append(f"成功将Prompt从{orig['token_count']} tokens优化到{opt['token_count']} tokens，"
                     f"减少了{results['reduction']['percent']:.1f}%，"
                     f"同时保持了语义的一致性。\n")

        return '\n'.join(report)

"""
作业3：智能模型选择器
实现一个智能系统，根据多个因素自动选择最佳模型和参数
"""

import tiktoken
from typing import Dict, List
from enum import Enum
import json

class TaskType(Enum):
    """任务类型枚举"""
    CREATIVE = "creative"        # 创意写作
    FACTUAL = "factual"          # 事实查询
    CODE = "code"                # 代码生成
    ANALYSIS = "analytical"      # 分析推理
    TRANSLATION = "translation"  # 翻译
    CONVERSATION = "conversation" # 对话

class ModelConfig:
    """模型配置类"""

    # 模型性能参数 (基于实际测试数据)
    MODEL_SPECS = {
        'gpt-3.5-turbo': {
            'cost_per_1k': 0.001,
            'avg_latency': 0.8,      # 秒
            'quality_score': 7.0,     # 1-10分
            'max_tokens': 4096,
            'best_for': [TaskType.CONVERSATION, TaskType.FACTUAL]
        },
        'gpt-4o-mini': {
            'cost_per_1k': 0.0006,
            'avg_latency': 1.2,
            'quality_score': 8.5,
            'max_tokens': 16384,
            'best_for': [TaskType.CODE, TaskType.ANALYSIS, TaskType.TRANSLATION]
        },
        'gpt-4o': {
            'cost_per_1k': 0.01,
            'avg_latency': 2.5,
            'quality_score': 9.5,
            'max_tokens': 128000,
            'best_for': [TaskType.CREATIVE, TaskType.ANALYSIS]
        }
    }

    # Temperature建议
    TEMPERATURE_GUIDE = {
        TaskType.CREATIVE: 0.9,
        TaskType.FACTUAL: 0.1,
        TaskType.CODE: 0.2,
        TaskType.ANALYSIS: 0.3,
        TaskType.TRANSLATION: 0.3,
        TaskType.CONVERSATION: 0.7
    }


def detect_task_type(prompt: str) -> TaskType:
    """检测任务类型"""

    prompt_lower = prompt.lower()

    # 关键词匹配
    creative_keywords = ['写', '创作', '诗', '故事', '想象', '设计']
    factual_keywords = ['是什么', '定义', '解释', '查询', '事实']
    code_keywords = ['代码', 'code', '函数', 'python', 'javascript', '编程', 'bug']
    analysis_keywords = ['分析', '推理', '为什么', '比较', '评估']
    translation_keywords = ['翻译', 'translate', '英文', '中文']

    if any(keyword in prompt_lower for keyword in creative_keywords):
        return TaskType.CREATIVE
    elif any(keyword in prompt_lower for keyword in code_keywords):
        return TaskType.CODE
    elif any(keyword in prompt_lower for keyword in analysis_keywords):
        return TaskType.ANALYSIS
    elif any(keyword in prompt_lower for keyword in translation_keywords):
        return TaskType.TRANSLATION
    elif any(keyword in prompt_lower for keyword in factual_keywords):
        return TaskType.FACTUAL
    else:
        return TaskType.CONVERSATION


def estimate_tokens(prompt: str) -> int:
    """估算prompt的token数量"""
    encoder = tiktoken.get_encoding("cl100k_base")
    return len(encoder.encode(prompt))


def smart_model_selector(
    prompt: str,
    max_budget: float,
    max_latency: float,
    quality_requirement: int  # 1-10
) -> dict:
    """
    智能模型选择器

    参数:
        prompt: 用户输入的prompt
        max_budget: 最大预算 (美元)
        max_latency: 最大延迟 (秒)
        quality_requirement: 质量要求 (1-10分)

    返回:
        {
            'model': 'gpt-3.5-turbo',
            'temperature': 0.7,
            'max_tokens': 150,
            'reasoning': '选择理由',
            'estimated_cost': 0.001,
            'estimated_latency': 0.8,
            'confidence': 0.9
        }
    """

    # 1. 检测任务类型
    task_type = detect_task_type(prompt)

    # 2. 估算token数量
    prompt_tokens = estimate_tokens(prompt)
    estimated_response_tokens = 200  # 默认估算
    total_tokens = prompt_tokens + estimated_response_tokens

    # 3. 筛选符合条件的模型
    suitable_models = []

    for model_name, specs in ModelConfig.MODEL_SPECS.items():
        # 计算预估成本
        estimated_cost = (total_tokens / 1000) * specs['cost_per_1k']

        # 检查是否符合预算
        if estimated_cost > max_budget:
            continue

        # 检查是否符合延迟要求
        if specs['avg_latency'] > max_latency:
            continue

        # 检查质量是否满足
        if specs['quality_score'] < quality_requirement:
            continue

        # 计算匹配度得分
        score = 0.0

        # 任务类型匹配度 (40%)
        if task_type in specs['best_for']:
            score += 40

        # 质量匹配度 (30%)
        quality_match = 1 - abs(specs['quality_score'] - quality_requirement) / 10
        score += quality_match * 30

        # 成本效率 (20%)
        cost_efficiency = 1 - (estimated_cost / max_budget)
        score += cost_efficiency * 20

        # 速度效率 (10%)
        speed_efficiency = 1 - (specs['avg_latency'] / max_latency)
        score += speed_efficiency * 10

        suitable_models.append({
            'model': model_name,
            'score': score,
            'estimated_cost': estimated_cost,
            'estimated_latency': specs['avg_latency'],
            'quality_score': specs['quality_score']
        })

    # 4. 如果没有符合条件的模型
    if not suitable_models:
        # 降低要求，选择最便宜的
        cheapest = min(
            ModelConfig.MODEL_SPECS.items(),
            key=lambda x: x[1]['cost_per_1k']
        )

        return {
            'model': cheapest[0],
            'temperature': ModelConfig.TEMPERATURE_GUIDE.get(task_type, 0.7),
            'max_tokens': 100,  # 减少token以降低成本
            'reasoning': f'⚠️ 无完全符合条件的模型。选择最经济的{cheapest[0]}，'
                        f'但可能超出预算或质量要求。建议放宽限制。',
            'estimated_cost': (total_tokens / 1000) * cheapest[1]['cost_per_1k'],
            'estimated_latency': cheapest[1]['avg_latency'],
            'confidence': 0.3
        }

    # 5. 选择得分最高的模型
    best_match = max(suitable_models, key=lambda x: x['score'])

    # 6. 确定temperature
    temperature = ModelConfig.TEMPERATURE_GUIDE.get(task_type, 0.7)

    # 7. 确定max_tokens
    # 根据预算动态调整
    remaining_budget = max_budget - best_match['estimated_cost']
    max_tokens = min(
        500,  # 最大值
        int((remaining_budget / ModelConfig.MODEL_SPECS[best_match['model']]['cost_per_1k']) * 1000)
    )
    max_tokens = max(50, max_tokens)  # 最小值

    # 8. 生成选择理由
    reasoning_parts = []

    reasoning_parts.append(f"✅ 任务类型: {task_type.value}")
    reasoning_parts.append(f"✅ 选择 {best_match['model']}")

    if task_type in ModelConfig.MODEL_SPECS[best_match['model']]['best_for']:
        reasoning_parts.append(f"  - 最适合{task_type.value}类任务")

    reasoning_parts.append(f"  - 质量评分: {best_match['quality_score']}/10 (要求: {quality_requirement}/10)")
    reasoning_parts.append(f"  - 预估成本: ${best_match['estimated_cost']:.6f} (预算: ${max_budget})")
    reasoning_parts.append(f"  - 预估延迟: {best_match['estimated_latency']}s (要求: <{max_latency}s)")
    reasoning_parts.append(f"  - Temperature设置: {temperature} (适合{task_type.value})")

    reasoning = '\n'.join(reasoning_parts)

    # 9. 计算置信度
    confidence = best_match['score'] / 100

    return {
        'model': best_match['model'],
        'temperature': temperature,
        'max_tokens': max_tokens,
        'reasoning': reasoning,
        'estimated_cost': best_match['estimated_cost'],
        'estimated_latency': best_match['estimated_latency'],
        'confidence': confidence,
        'task_type': task_type.value
    }

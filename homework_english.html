# model_comparison_homework.py
"""
Assignment 1: Model Comparison Report
Business Scenario: Smart Customer Service Dialogue System
Comparison Models: GPT-3.5-turbo, GPT-4o-mini, GPT-4o
"""

import openai
import time
import pandas as pd
from datetime import datetime
from typing import List, Dict
import json

class ModelComparisonReport:
    """Model Comparison Testing and Report Generation System"""

    def __init__(self, api_key: str):
        self.client = openai.OpenAI(api_key=api_key)
        self.results = []

    def test_model(self,
                   model: str,
                   prompt: str,
                   temperature: float = 0.7) -> Dict:
        """Test single model response"""

        start_time = time.time()

        try:
            response = self.client.chat.completions.create(
                model=model,
                messages=[
                    {"role": "system", "content": "You are a professional customer service representative"},
                    {"role": "user", "content": prompt}
                ],
                temperature=temperature,
                max_tokens=200
            )

            elapsed_time = time.time() - start_time

            return {
                'success': True,
                'response': response.choices[0].message.content,
                'tokens': response.usage.total_tokens,
                'prompt_tokens': response.usage.prompt_tokens,
                'completion_tokens': response.usage.completion_tokens,
                'time': elapsed_time,
                'cost': self._calculate_cost(model, response.usage.total_tokens)
            }

        except Exception as e:
            return {
                'success': False,
                'error': str(e),
                'tokens': 0,
                'time': 0,
                'cost': 0
            }

    def _calculate_cost(self, model: str, tokens: int) -> float:
        """Calculate API call cost"""
        pricing = {
            'gpt-3.5-turbo': 0.001,      # $0.001 per 1K tokens
            'gpt-4o-mini': 0.0006,       # $0.0006 per 1K tokens
            'gpt-4o': 0.01,              # $0.01 per 1K tokens
        }
        price_per_1k = pricing.get(model, 0.001)
        return (tokens / 1000) * price_per_1k

    def run_comparison(self,
                      models: List[str],
                      test_prompts: List[str],
                      temperature: float = 0.7):
        """Run complete comparison test"""

        print("🚀 Starting Model Comparison Test")
        print("=" * 70)

        for model in models:
            print(f"\n📊 Testing Model: {model}")
            print("-" * 70)

            for idx, prompt in enumerate(test_prompts, 1):
                print(f"\n  Test {idx}/{len(test_prompts)}: {prompt[:40]}...")

                result = self.test_model(model, prompt, temperature)

                if result['success']:
                    print(f"    ✓ Success")
                    print(f"    ⏱️  Response time: {result['time']:.2f}s")
                    print(f"    🔤 Token count: {result['tokens']}")
                    print(f"    💰 Cost: ${result['cost']:.6f}")

                    # Evaluate response quality (1-10 scale)
                    quality_score = self._evaluate_quality(
                        prompt,
                        result['response']
                    )
                    print(f"    ⭐ Quality score: {quality_score}/10")

                    # Save results
                    self.results.append({
                        'model': model,
                        'prompt': prompt,
                        'response': result['response'],
                        'quality_score': quality_score,
                        'response_time': result['time'],
                        'total_tokens': result['tokens'],
                        'prompt_tokens': result['prompt_tokens'],
                        'completion_tokens': result['completion_tokens'],
                        'cost': result['cost'],
                        'timestamp': datetime.now().isoformat()
                    })
                else:
                    print(f"    ✗ Failed: {result['error']}")

                # Avoid rate limiting
                time.sleep(1)

    def _evaluate_quality(self, prompt: str, response: str) -> float:
        """Evaluate response quality (simplified version)"""
        score = 10.0

        # Check response length
        if len(response) < 20:
            score -= 3
        elif len(response) > 500:
            score -= 1

        # Check for polite language
        polite_words = ['you', 'please', 'thank', 'appreciate', 'sorry']
        if not any(word in response.lower() for word in polite_words):
            score -= 1

        # Check for structured content
        if ':' in response or '.' in response:
            score += 0.5

        return min(max(score, 0), 10)

    def generate_report(self) -> str:
        """Generate Markdown format comparison report"""

        if not self.results:
            return "# Error: No test results\n"

        df = pd.DataFrame(self.results)

        report = []
        report.append("# Model Comparison Test Report\n")
        report.append(f"**Test Time**: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
        report.append(f"**Business Scenario**: Smart Customer Service Dialogue System\n")
        report.append(f"**Models Tested**: {df['model'].nunique()}\n")
        report.append(f"**Test Samples**: {len(df)}\n\n")

        # 1. Overall comparison table
        report.append("## 1. Overall Model Performance Comparison\n\n")

        summary = df.groupby('model').agg({
            'quality_score': 'mean',
            'response_time': 'mean',
            'total_tokens': 'mean',
            'cost': 'sum'
        }).round(4)

        report.append("| Model | Avg Quality | Avg Response Time(s) | Avg Tokens | Total Cost($) |")
        report.append("|-------|-------------|---------------------|------------|---------------|")

        for model, row in summary.iterrows():
            report.append(
                f"| {model} | {row['quality_score']:.2f} | "
                f"{row['response_time']:.2f} | {row['total_tokens']:.0f} | "
                f"${row['cost']:.6f} |"
            )

        report.append("\n")

        # 2. Detailed test results
        report.append("## 2. Detailed Test Results\n\n")

        for model in df['model'].unique():
            report.append(f"### {model}\n\n")

            model_df = df[df['model'] == model]

            for idx, row in model_df.iterrows():
                report.append(f"**Test {idx + 1}**\n")
                report.append(f"- **Prompt**: {row['prompt']}\n")
                report.append(f"- **Response**: {row['response'][:100]}...\n")
                report.append(f"- **Quality Score**: {row['quality_score']}/10\n")
                report.append(f"- **Response Time**: {row['response_time']:.2f}s\n")
                report.append(f"- **Token Usage**: {row['total_tokens']} "
                            f"(input: {row['prompt_tokens']}, "
                            f"output: {row['completion_tokens']})\n")
                report.append(f"- **Cost**: ${row['cost']:.6f}\n\n")

        # 3. Performance analysis
        report.append("## 3. Performance Analysis\n\n")

        # Find best models
        best_quality = summary['quality_score'].idxmax()
        best_speed = summary['response_time'].idxmin()
        best_cost = summary['cost'].idxmin()

        report.append(f"- **Best Quality**: {best_quality} "
                     f"(avg score: {summary.loc[best_quality, 'quality_score']:.2f})\n")
        report.append(f"- **Fastest**: {best_speed} "
                     f"(avg response: {summary.loc[best_speed, 'response_time']:.2f}s)\n")
        report.append(f"- **Lowest Cost**: {best_cost} "
                     f"(total cost: ${summary.loc[best_cost, 'cost']:.6f})\n\n")

        # 4. Conclusions and recommendations
        report.append("## 4. Conclusions and Recommendations\n\n")
        report.append("### Comprehensive Analysis\n\n")

        # Calculate cost efficiency (quality/cost)
        summary['cost_efficiency'] = summary['quality_score'] / (summary['cost'] * 1000)
        best_efficiency = summary['cost_efficiency'].idxmax()

        report.append(f"Based on this test, **{best_efficiency}** "
                     f"shows the best cost-effectiveness for customer service scenarios.\n\n")

        report.append("### Usage Recommendations\n\n")
        report.append("1. **High-frequency simple queries**: Use GPT-3.5-turbo (low cost, fast response)\n")
        report.append("2. **Complex issue handling**: Use GPT-4o (high quality, strong understanding)\n")
        report.append("3. **Balance performance and cost**: Use GPT-4o-mini (best overall value)\n")
        report.append("4. **Cost optimization strategy**: Combining FAQ system + smart routing can save 90%+ costs\n\n")

        # 5. Appendix
        report.append("## 5. Appendix\n\n")
        report.append("### Test Environment\n")
        report.append("- Python version: 3.11+\n")
        report.append("- OpenAI library version: 1.12.0\n")
        report.append("- Temperature: 0.7\n")
        report.append("- Max Tokens: 200\n\n")

        return '\n'.join(report)

    def save_report(self, filename: str = None):
        """Save report to file"""
        if filename is None:
            filename = f"model_comparison_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.md"

        report_content = self.generate_report()

        with open(filename, 'w', encoding='utf-8') as f:
            f.write(report_content)

        print(f"\n✅ Report saved: {filename}")

        # Also save raw data as CSV
        csv_filename = filename.replace('.md', '.csv')
        df = pd.DataFrame(self.results)
        df.to_csv(csv_filename, index=False, encoding='utf-8-sig')

        print(f"✅ Raw data saved: {csv_filename}")


# Main program
if __name__ == "__main__":

# token_optimization_homework.py
"""
Assignment 2: Token Optimization Practice
Goal: Optimize original prompt from 200+ tokens to <100 tokens while maintaining semantics
"""

import openai
import tiktoken
from typing import Dict, List
import pandas as pd

class TokenOptimizer:
    """Token Optimization Tool Class"""

    def __init__(self, api_key: str):
        self.client = openai.OpenAI(api_key=api_key)
        self.encoder = tiktoken.get_encoding("cl100k_base")

    def count_tokens(self, text: str) -> int:
        """Calculate text token count"""
        return len(self.encoder.encode(text))

    def analyze_prompt(self, prompt: str) -> Dict:
        """Analyze prompt token usage"""
        tokens = self.encoder.encode(prompt)
        token_count = len(tokens)

        # Token breakdown
        token_breakdown = []
        for token in tokens[:20]:  # Only show first 20
            decoded = self.encoder.decode([token])
            token_breakdown.append(decoded)

        return {
            'text': prompt,
            'token_count': token_count,
            'char_count': len(prompt),
            'token_breakdown': token_breakdown,
            'efficiency': len(prompt) / token_count if token_count > 0 else 0
        }

    def test_prompt_quality(self, prompt: str, test_case: str) -> Dict:
        """Test prompt response quality"""
        try:
            response = self.client.chat.completions.create(
                model="gpt-3.5-turbo",
                messages=[
                    {"role": "system", "content": prompt},
                    {"role": "user", "content": test_case}
                ],
                temperature=0.7,
                max_tokens=150
            )

            return {
                'success': True,
                'response': response.choices[0].message.content,
                'tokens_used': response.usage.total_tokens,
                'cost': (response.usage.total_tokens / 1000) * 0.001
            }
        except Exception as e:
            return {
                'success': False,
                'error': str(e)
            }

    def compare_prompts(self,
                       original: str,
                       optimized: str,
                       test_cases: List[str]) -> Dict:
        """Compare original and optimized prompts"""

        print("🔬 Starting Token Optimization Comparison Test")
        print("=" * 70)

        # Analyze token usage
        print("\n📊 Token Analysis")
        print("-" * 70)

        original_analysis = self.analyze_prompt(original)
        optimized_analysis = self.analyze_prompt(optimized)

        print(f"\nOriginal Prompt:")
        print(f"  Text: {original[:60]}...")
        print(f"  Characters: {original_analysis['char_count']}")
        print(f"  Tokens: {original_analysis['token_count']}")

        print(f"\nOptimized Prompt:")
        print(f"  Text: {optimized[:60]}...")
        print(f"  Characters: {optimized_analysis['char_count']}")
        print(f"  Tokens: {optimized_analysis['token_count']}")

        # Calculate optimization effect
        token_reduction = original_analysis['token_count'] - optimized_analysis['token_count']
        reduction_percent = (token_reduction / original_analysis['token_count']) * 100

        print(f"\n✨ Optimization Effect:")
        print(f"  Tokens reduced: {token_reduction}")
        print(f"  Reduction ratio: {reduction_percent:.1f}%")

        # Quality comparison test
        print("\n🎯 Quality Comparison Test")
        print("-" * 70)

        results = {
            'original_analysis': original_analysis,
            'optimized_analysis': optimized_analysis,
            'reduction': {
                'tokens': token_reduction,
                'percent': reduction_percent
            },
            'quality_tests': []
        }

        for idx, test_case in enumerate(test_cases, 1):
            print(f"\nTest case {idx}: {test_case[:40]}...")

            # Test original prompt
            print("  Original Prompt response...", end="")
            original_result = self.test_prompt_quality(original, test_case)
            if original_result['success']:
                print(f" ✓ ({original_result['tokens_used']} tokens)")
            else:
                print(f" ✗ Failed")

            # Test optimized prompt
            print("  Optimized Prompt response...", end="")
            optimized_result = self.test_prompt_quality(optimized, test_case)
            if optimized_result['success']:
                print(f" ✓ ({optimized_result['tokens_used']} tokens)")
            else:
                print(f" ✗ Failed")

            # Compare results
            if original_result['success'] and optimized_result['success']:
                quality_score = self._compare_quality(
                    original_result['response'],
                    optimized_result['response']
                )

                token_saving = original_result['tokens_used'] - optimized_result['tokens_used']
                cost_saving = original_result['cost'] - optimized_result['cost']

                print(f"  Semantic similarity: {quality_score:.1f}%")
                print(f"  Tokens saved: {token_saving}")
                print(f"  Cost saved: ${cost_saving:.6f}")

                results['quality_tests'].append({
                    'test_case': test_case,
                    'original_response': original_result['response'],
                    'optimized_response': optimized_result['response'],
                    'quality_score': quality_score,
                    'token_saving': token_saving,
                    'cost_saving': cost_saving
                })

        return results

    def _compare_quality(self, response1: str, response2: str) -> float:
        """Simple semantic similarity assessment"""
        # Simplified version: based on length and keyword overlap
        words1 = set(response1.lower().split())
        words2 = set(response2.lower().split())

        overlap = len(words1 & words2)
        total = len(words1 | words2)

        similarity = (overlap / total * 100) if total > 0 else 0
        return similarity

    def generate_optimization_report(self, results: Dict) -> str:
        """Generate optimization report"""
        report = []

        report.append("# Token Optimization Practice Report\n")
        report.append("## 1. Before and After Comparison\n")

        orig = results['original_analysis']
        opt = results['optimized_analysis']

        report.append("| Metric | Original Prompt | Optimized Prompt | Improvement |")
        report.append("|--------|-----------------|------------------|-------------|")
        report.append(f"| Tokens | {orig['token_count']} | {opt['token_count']} | "
                     f"-{results['reduction']['tokens']} ({results['reduction']['percent']:.1f}%) |")
        report.append(f"| Characters | {orig['char_count']} | {opt['char_count']} | "
                     f"-{orig['char_count'] - opt['char_count']} |")

        report.append("\n## 2. Original Prompt\n")
        report.append(f"```\n{orig['text']}\n```\n")

        report.append("## 3. Optimized Prompt\n")
        report.append(f"```\n{opt['text']}\n```\n")

        report.append("## 4. Optimization Strategies\n")
        report.append("1. Removed redundant words and repetitive expressions\n")
        report.append("2. Used more concise sentence structures\n")
        report.append("3. Retained core instructions and key information\n")
        report.append("4. Removed unnecessary polite phrases\n\n")

        report.append("## 5. Quality Validation\n")

        if results['quality_tests']:
            avg_similarity = sum(t['quality_score'] for t in results['quality_tests']) / len(results['quality_tests'])
            total_token_saving = sum(t['token_saving'] for t in results['quality_tests'])

            report.append(f"- Average semantic similarity: {avg_similarity:.1f}%\n")
            report.append(f"- Total tokens saved: {total_token_saving}\n")
            report.append(f"- Test cases: {len(results['quality_tests'])}\n\n")

        report.append("## 6. Conclusion\n")
        report.append(f"Successfully optimized prompt from {orig['token_count']} tokens to {opt['token_count']} tokens, "
                     f"reducing by {results['reduction']['percent']:.1f}%, "
                     f"while maintaining semantic consistency.\n")

        return '\n'.join(report)

def detect_task_type(prompt: str) -> TaskType:
    """Detect task type"""

    prompt_lower = prompt.lower()

    # Keyword matching
    creative_keywords = ['write', 'create', 'poem', 'story', 'imagine', 'design']
    factual_keywords = ['what is', 'define', 'explain', 'query', 'fact']
    code_keywords = ['code', 'function', 'python', 'javascript', 'programming', 'bug']
    analysis_keywords = ['analyze', 'reasoning', 'why', 'compare', 'evaluate']
    translation_keywords = ['translate', 'english', 'chinese']

    if any(keyword in prompt_lower for keyword in creative_keywords):
        return TaskType.CREATIVE
    elif any(keyword in prompt_lower for keyword in code_keywords):
        return TaskType.CODE
    elif any(keyword in prompt_lower for keyword in analysis_keywords):
        return TaskType.ANALYSIS
    elif any(keyword in prompt_lower for keyword in translation_keywords):
        return TaskType.TRANSLATION
    elif any(keyword in prompt_lower for keyword in factual_keywords):
        return TaskType.FACTUAL
    else:
        return TaskType.CONVERSATION


def estimate_tokens(prompt: str) -> int:

    """
    Smart Model Selector

    Parameters:
        prompt: User input prompt
        max_budget: Maximum budget (USD)
        max_latency: Maximum latency (seconds)
        quality_requirement: Quality requirement (1-10 scale)

    Returns:
        {
            'model': 'gpt-3.5-turbo',
            'temperature': 0.7,
            'max_tokens': 150,
            'reasoning': 'Selection reasoning',
            'estimated_cost': 0.001,
            'estimated_latency': 0.8,
            'confidence': 0.9
        }
    """

    # 1. Detect task type
    task_type = detect_task_type(prompt)

    # 2. Estimate token count
    prompt_tokens = estimate_tokens(prompt)
    estimated_response_tokens = 200  # Default estimate
    total_tokens = prompt_tokens + estimated_response_tokens

    # 3. Filter suitable models
    suitable_models = []

    for model_name, specs in ModelConfig.MODEL_SPECS.items():
        # Calculate estimated cost
        estimated_cost = (total_tokens / 1000) * specs['cost_per_1k']

        # Check budget compliance
        if estimated_cost > max_budget:
            continue

        # Check latency compliance
        if specs['avg_latency'] > max_latency:
            continue

        # Check quality satisfaction
        if specs['quality_score'] < quality_requirement:
            continue

        # Calculate match score
        score = 0.0

        # Task type match (40%)
        if task_type in specs['best_for']:
            score += 40

        # Quality match (30%)
        quality_match = 1 - abs(specs['quality_score'] - quality_requirement) / 10
        score += quality_match * 30

        # Cost efficiency (20%)
        cost_efficiency = 1 - (estimated_cost / max_budget)
        score += cost_efficiency * 20

        # Speed efficiency (10%)
        speed_efficiency = 1 - (specs['avg_latency'] / max_latency)
        score += speed_efficiency * 10

        suitable_models.append({
            'model': model_name,
            'score': score,
            'estimated_cost': estimated_cost,
            'estimated_latency': specs['avg_latency'],
            'quality_score': specs['quality_score']
        })

    # 4. If no suitable models
    if not suitable_models:
        # Lower requirements, choose cheapest
        cheapest = min(
            ModelConfig.MODEL_SPECS.items(),
            key=lambda x: x[1]['cost_per_1k']
        )

        return {
            'model': cheapest[0],
            'temperature': ModelConfig.TEMPERATURE_GUIDE.get(task_type, 0.7),
            'max_tokens': 100,  # Reduce tokens to lower cost
            'reasoning': f'⚠️ No fully compliant models. Selected the most economical {cheapest[0]}, '
                        f'but may exceed budget or quality requirements. Consider relaxing constraints.',
            'estimated_cost': (total_tokens / 1000) * cheapest[1]['cost_per_1k'],
            'estimated_latency': cheapest[1]['avg_latency'],
            'confidence': 0.3
        }

    # 5. Select highest scoring model
    best_match = max(suitable_models, key=lambda x: x['score'])

    # 6. Determine temperature
    temperature = ModelConfig.TEMPERATURE_GUIDE.get(task_type, 0.7)

    # 7. Determine max_tokens
    # Dynamically adjust based on budget
    remaining_budget = max_budget - best_match['estimated_cost']
    max_tokens = min(
        500,  # Maximum value
        int((remaining_budget / ModelConfig.MODEL_SPECS[best_match['model']]['cost_per_1k']) * 1000)
    )
    max_tokens = max(50, max_tokens)  # Minimum value

    # 8. Generate selection reasoning
    reasoning_parts = []

    reasoning_parts.append(f"✅ Task type: {task_type.value}")
    reasoning_parts.append(f"✅ Selected {best_match['model']}")

    if task_type in ModelConfig.MODEL_SPECS[best_match['model']]['best_for']:
        reasoning_parts.append(f"  - Best suited for {task_type.value} tasks")

    reasoning_parts.append(f"  - Quality score: {best_match['quality_score']}/10 (requirement: {quality_requirement}/10)")
    reasoning_parts.append(f"  - Estimated cost: ${best_match['estimated_cost']:.6f} (budget: ${max_budget})")
    reasoning_parts.append(f"  - Estimated latency: {best_match['estimated_latency']}s (requirement: <{max_latency}s)")
    reasoning_parts.append(f"  - Temperature setting: {temperature} (suitable for {task_type.value})")

    reasoning = '\n'.join(reasoning_parts)

    # 9. Calculate confidence
    confidence = best_match['score'] / 100

    return {
        'model': best_match['model'],
        'temperature': temperature,
        'max_tokens': max_tokens,
        'reasoning': reasoning,
        'estimated_cost': best_match['estimated_cost'],
        'estimated_latency': best_match['estimated_latency'],
        'confidence': confidence,
        'task_type': task_type.value
    }

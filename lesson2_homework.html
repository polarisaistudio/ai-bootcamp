        return True

1. ChatManager 类

"""聊天管理器"""
import time
from typing import List, Dict, Optional
from openai import OpenAI, APIError, RateLimitError, APIConnectionError
from config import Config

class ChatManager:
    """管理OpenAI聊天对话"""

    def __init__(self, model: str = None):
        """初始化聊天管理器"""
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = model or Config.DEFAULT_MODEL
        self.conversation_history: List[Dict] = []
        self.total_tokens = 0

    def add_message(self, role: str, content: str):
        """添加消息到对话历史"""
        self.conversation_history.append({
            "role": role,
            "content": content
        })

    def chat(self, user_input: str, **kwargs) -> Optional[str]:
        """发送聊天请求并返回响应"""
        # 添加用户消息
        self.add_message("user", user_input)

        # 尝试发送请求（带重试）
        for attempt in range(Config.MAX_RETRIES):
            try:
                response = self.client.chat.completions.create(
                    model=self.model,
                    messages=self.conversation_history,
                    temperature=kwargs.get('temperature', Config.TEMPERATURE),
                    max_tokens=kwargs.get('max_tokens', Config.MAX_TOKENS)
                )

                # 提取响应
                assistant_message = response.choices[0].message.content
                self.add_message("assistant", assistant_message)

                # 更新令牌统计
                self.total_tokens += response.usage.total_tokens

                return assistant_message

            except RateLimitError:
                if attempt < Config.MAX_RETRIES - 1:
                    wait_time = Config.RETRY_DELAY * (2 ** attempt)
                    print(f"⚠️  速率限制，{wait_time}秒后重试...")
                    time.sleep(wait_time)
                else:
                    print("❌ 达到速率限制，请稍后再试")
                    return None

            except APIConnectionError:
                if attempt < Config.MAX_RETRIES - 1:
                    wait_time = Config.RETRY_DELAY * (2 ** attempt)
                    print(f"⚠️  网络错误，{wait_time}秒后重试...")
                    time.sleep(wait_time)
                else:
                    print("❌ 网络连接失败，请检查网络")
                    return None

            except APIError as e:
                print(f"❌ API错误：{e}")
                return None

            except Exception as e:
                print(f"❌ 未知错误：{e}")
                return None

        return None

    def clear_history(self):
        """清空对话历史"""
        self.conversation_history = []
        self.total_tokens = 0

    def get_stats(self) -> Dict:
        """获取对话统计"""
        return {
            "rounds": len(self.conversation_history) // 2,
            "messages": len(self.conversation_history),
            "total_tokens": self.total_tokens,
            "model": self.model
        }

    stats = manager.get_stats()
    stats_text = f"""
## 📊 对话统计

- 对话轮次：{stats['rounds']}
- 消息总数：{stats['messages']}
- 令牌消耗：{stats['total_tokens']}
- 使用模型：{stats['model']}
    """
    console.print(Panel(Markdown(stats_text), border_style="yellow"))

def main():
    """主程序"""
    # 验证配置
    try:
        Config.validate()
    except ValueError as e:
        console.print(f"[red]{e}[/red]")
        return

    # 显示欢迎信息
    show_welcome()

    # 创建聊天管理器
    manager = ChatManager()

    # 主循环
    while True:
        try:
            # 获取用户输入
            user_input = Prompt.ask("\n[bold cyan]你[/bold cyan]")

            # 处理空输入
            if not user_input.strip():
                continue

            # 处理命令
            if user_input.startswith('/'):
                command = user_input.lower().strip()

                if command == '/quit' or command == '/exit':
                    console.print("[yellow]再见！👋[/yellow]")
                    break

                elif command == '/help':
                    show_help()
                    continue

                elif command == '/clear':
                    manager.clear_history()
                    console.print("[green]✅ 对话历史已清空[/green]")
                    continue

                elif command == '/stats':
                    show_stats(manager)
                    continue

                else:
                    console.print(f"[red]未知命令：{command}[/red]")
                    console.print("[yellow]输入 /help 查看可用命令[/yellow]")
                    continue

            # 发送聊天请求
            with console.status("[bold green]AI思考中..."):
                response = manager.chat(user_input)

            # 显示响应
            if response:
                console.print(Panel(
                    Markdown(response),
                    title="[bold green]🤖 AI助手[/bold green]",
                    border_style="green"
                ))
            else:
                console.print("[red]抱歉，无法获取响应，请重试[/red]")

        except KeyboardInterrupt:
            console.print("\n[yellow]检测到中断，退出中...👋[/yellow]")
            break

        except Exception as e:
            console.print(f"[red]错误：{e}[/red]")

if __name__ == "__main__":
    main()

1. 流式响应功能

"""流式聊天功能"""
import sys
import time
from openai import OpenAI
from config import Config

class StreamChatManager:
    """支持流式输出的聊天管理器"""

    def __init__(self, model: str = None):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = model or Config.DEFAULT_MODEL
        self.conversation_history = []

    def chat_stream(self, user_input: str, **kwargs):
        """流式聊天"""
        self.conversation_history.append({
            "role": "user",
            "content": user_input
        })

        try:
            stream = self.client.chat.completions.create(
                model=self.model,
                messages=self.conversation_history,
                temperature=kwargs.get('temperature', Config.TEMPERATURE),
                max_tokens=kwargs.get('max_tokens', Config.MAX_TOKENS),
                stream=True
            )

            full_response = ""
            print("\n🤖 AI助手: ", end="", flush=True)

            for chunk in stream:
                if chunk.choices[0].delta.content:
                    content = chunk.choices[0].delta.content
                    print(content, end="", flush=True)
                    full_response += content
                    time.sleep(0.01)  # 打字机效果

            print("\n")  # 换行

            # 保存完整响应
            self.conversation_history.append({
                "role": "assistant",
                "content": full_response
            })

            return full_response

        except Exception as e:
            print(f"\n❌ 错误：{e}")
            return None

"""对话导出功能"""
import os
from datetime import datetime
from typing import List, Dict

class ConversationExporter:
    """对话导出工具"""

    @staticmethod
    def export_to_markdown(
        conversation: List[Dict],
        filepath: str = None,
        stats: Dict = None
    ) -> str:
        """导出对话为Markdown格式"""

        # 生成文件名
        if not filepath:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filepath = f"conversation_{timestamp}.md"

        # 构建Markdown内容
        lines = []
        lines.append("# AI对话记录\n")
        lines.append(f"**导出时间：** {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")

        # 添加统计信息
        if stats:
            lines.append("## 📊 对话统计\n")
            lines.append(f"- 对话轮次：{stats.get('rounds', 0)}")
            lines.append(f"- 消息总数：{stats.get('messages', 0)}")
            lines.append(f"- 令牌消耗：{stats.get('total_tokens', 0)}")
            lines.append(f"- 使用模型：{stats.get('model', 'N/A')}\n")

        lines.append("## 💬 对话内容\n")

        # 添加对话
        for i, msg in enumerate(conversation, 1):
            role = msg['role']
            content = msg['content']

            if role == 'user':
                lines.append(f"### 👤 用户 (#{i//2 + 1})\n")
            else:
                lines.append(f"### 🤖 AI助手 (#{i//2})\n")

            lines.append(f"{content}\n")

        # 写入文件
        content = "\n".join(lines)
        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(content)

        return filepath

# 使用示例
def export_conversation(manager):
    """导出对话"""
    exporter = ConversationExporter()
    filepath = exporter.export_to_markdown(
        conversation=manager.conversation_history,
        stats=manager.get_stats()
    )
    print(f"✅ 对话已导出到：{filepath}")

from typing import Dict, List
from rich.console import Console
from rich.table import Table

class ModelManager:
    """模型管理器"""

    MODELS = {
        "gpt-3.5-turbo": {
            "name": "GPT-3.5 Turbo",
            "description": "快速、经济的选择",
            "input_price": "$0.0005/1K tokens",
            "output_price": "$0.0015/1K tokens",
            "max_tokens": 4096
        },
        "gpt-4": {
            "name": "GPT-4",
            "description": "最强大的模型",
            "input_price": "$0.03/1K tokens",
            "output_price": "$0.06/1K tokens",
            "max_tokens": 8192
        },
        "gpt-4-turbo-preview": {
            "name": "GPT-4 Turbo",
            "description": "GPT-4的优化版本",
            "input_price": "$0.01/1K tokens",
            "output_price": "$0.03/1K tokens",
            "max_tokens": 128000
        }
    }

    @classmethod
    def list_models(cls):
        """列出所有可用模型"""
        console = Console()
        table = Table(title="🤖 可用模型")

        table.add_column("模型ID", style="cyan")
        table.add_column("名称", style="green")
        table.add_column("描述", style="yellow")
        table.add_column("输入价格", style="magenta")
        table.add_column("输出价格", style="magenta")

        for model_id, info in cls.MODELS.items():
            table.add_row(
                model_id,
                info["name"],
                info["description"],
                info["input_price"],
                info["output_price"]
            )

        console.print(table)

    @classmethod
    def get_model_info(cls, model_id: str) -> Dict:
        """获取模型信息"""
        return cls.MODELS.get(model_id, None)

    @classmethod
    def validate_model(cls, model_id: str) -> bool:
        """验证模型是否可用"""
        return model_id in cls.MODELS

# 在CLI中添加模型切换命令
def handle_model_command(manager, args):
    """处理模型相关命令"""
    if not args or args == 'list':
        ModelManager.list_models()
    else:
        model_id = args.strip()
        if ModelManager.validate_model(model_id):
            manager.model = model_id
            info = ModelManager.get_model_info(model_id)
            console.print(f"[green]✅ 已切换到模型：{info['name']}[/green]")
        else:
            console.print(f"[red]❌ 未知模型：{model_id}[/red]")
            console.print("[yellow]使用 /model list 查看可用模型[/yellow]")

"""增强版CLI - 集成所有高级功能"""
from rich.console import Console
from rich.panel import Panel
from rich.markdown import Markdown
from rich.prompt import Prompt, Confirm
from stream_chat import StreamChatManager
from exporter import ConversationExporter
from model_manager import ModelManager

console = Console()

def show_advanced_help():
    """显示高级功能帮助"""
    help_text = """
## 📖 高级功能指南

### 基础命令
- `/help` - 显示帮助
- `/quit` - 退出程序

### 对话管理
- `/clear` - 清空对话历史
- `/stats` - 显示统计信息
- `/export` - 导出对话记录

### 模型管理
- `/model list` - 列出所有模型
- `/model ` - 切换模型

### 高级设置
- `/stream on/off` - 开启/关闭流式输出
- `/temperature ` - 设置温度参数(0-2)
    """
    console.print(Panel(Markdown(help_text), border_style="blue"))

def main():
    """增强版主程序"""
    # 初始化
    ModelManager.list_models()
    manager = StreamChatManager()
    exporter = ConversationExporter()
    stream_enabled = True

    console.print("[bold green]🚀 欢迎使用增强版AI聊天工具！[/bold green]")
    console.print("[yellow]输入 /help 查看所有功能[/yellow]\n")

    while True:
        try:
            user_input = Prompt.ask("\n[bold cyan]你[/bold cyan]")

            if not user_input.strip():
                continue

            # 处理命令
            if user_input.startswith('/'):
                parts = user_input.split(maxsplit=1)
                command = parts[0].lower()
                args = parts[1] if len(parts) > 1 else ""

                if command in ['/quit', '/exit']:
                    if Confirm.ask("确定要退出吗？"):
                        console.print("[yellow]再见！👋[/yellow]")
                        break
                    continue

                elif command == '/help':
                    show_advanced_help()
                    continue

                elif command == '/export':
                    filepath = exporter.export_to_markdown(
                        manager.conversation_history,
                        stats=manager.get_stats()
                    )
                    console.print(f"[green]✅ 已导出到：{filepath}[/green]")
                    continue

                elif command == '/model':
                    handle_model_command(manager, args)
                    continue

                elif command == '/stream':
                    if args.lower() == 'on':
                        stream_enabled = True
                        console.print("[green]✅ 流式输出已开启[/green]")
                    elif args.lower() == 'off':
                        stream_enabled = False
                        console.print("[yellow]流式输出已关闭[/yellow]")
                    continue

            # 发送请求
            if stream_enabled:
                manager.chat_stream(user_input)
            else:
                with console.status("[bold green]AI思考中..."):
                    response = manager.chat(user_input)
                if response:
                    console.print(Panel(
                        Markdown(response),
                        title="[bold green]🤖 AI助手[/bold green]",
                        border_style="green"
                    ))

        except KeyboardInterrupt:
            console.print("\n[yellow]退出中...👋[/yellow]")
            break
        except Exception as e:
            console.print(f"[red]错误：{e}[/red]")

if __name__ == "__main__":
    main()

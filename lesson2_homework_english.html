"""Configuration management module"""
import os
from dotenv import load_dotenv

# Load environment variables
load_dotenv()

class Config:
    """Centralized application configuration management"""

    # API configuration
    OPENAI_API_KEY = os.getenv('OPENAI_API_KEY')
    OPENAI_ORG_ID = os.getenv('OPENAI_ORG_ID')

    # Model configuration
    DEFAULT_MODEL = "gpt-3.5-turbo"
    MAX_TOKENS = 1000
    TEMPERATURE = 0.7

    # Error handling configuration
    MAX_RETRIES = 3
    RETRY_DELAY = 1  # seconds

    @classmethod
    def validate(cls):
        """Validate configuration completeness"""
        if not cls.OPENAI_API_KEY:
            raise ValueError("❌ Error: OPENAI_API_KEY not set")

        if not cls.OPENAI_API_KEY.startswith('sk-'):
            raise ValueError("❌ Error: Invalid API key format")

        print("✅ Configuration validated successfully")
        return True

# Test configuration
if __name__ == "__main__":

    print(f"Default model: {Config.DEFAULT_MODEL}")
    print(f"Max tokens: {Config.MAX_TOKENS}")

1. ChatManager Class

"""Chat manager"""
import time
from typing import List, Dict, Optional
from openai import OpenAI, APIError, RateLimitError, APIConnectionError
from config import Config

class ChatManager:
    """Manage OpenAI chat conversations"""

    def __init__(self, model: str = None):
        """Initialize chat manager"""
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = model or Config.DEFAULT_MODEL
        self.conversation_history: List[Dict] = []
        self.total_tokens = 0

    def add_message(self, role: str, content: str):
        """Add message to conversation history"""
        self.conversation_history.append({
            "role": role,
            "content": content
        })

    def chat(self, user_input: str, **kwargs) -> Optional[str]:
        """Send chat request and return response"""
        # Add user message
        self.add_message("user", user_input)

        # Attempt to send request (with retry)
        for attempt in range(Config.MAX_RETRIES):
            try:
                response = self.client.chat.completions.create(
                    model=self.model,
                    messages=self.conversation_history,
                    temperature=kwargs.get('temperature', Config.TEMPERATURE),
                    max_tokens=kwargs.get('max_tokens', Config.MAX_TOKENS)
                )

                # Extract response
                assistant_message = response.choices[0].message.content
                self.add_message("assistant", assistant_message)

                # Update token statistics
                self.total_tokens += response.usage.total_tokens

                return assistant_message

            except RateLimitError:
                if attempt < Config.MAX_RETRIES - 1:
                    wait_time = Config.RETRY_DELAY * (2 ** attempt)
                    print(f"⚠️  Rate limit reached, retrying in {wait_time}s...")
                    time.sleep(wait_time)
                else:
                    print("❌ Rate limit reached, please try again later")
                    return None

            except APIConnectionError:
                if attempt < Config.MAX_RETRIES - 1:
                    wait_time = Config.RETRY_DELAY * (2 ** attempt)
                    print(f"⚠️  Network error, retrying in {wait_time}s...")
                    time.sleep(wait_time)
                else:
                    print("❌ Network connection failed, please check your network")
                    return None

            except APIError as e:
                print(f"❌ API error: {e}")
                return None

            except Exception as e:
                print(f"❌ Unknown error: {e}")
                return None

        return None

    def clear_history(self):
        """Clear conversation history"""
        self.conversation_history = []
        self.total_tokens = 0

    def get_stats(self) -> Dict:
        """Get conversation statistics"""
        return {
            "rounds": len(self.conversation_history) // 2,
            "messages": len(self.conversation_history),
            "total_tokens": self.total_tokens,
            "model": self.model
        }

"""CLI chat tool main program"""
from rich.console import Console
from rich.panel import Panel
from rich.markdown import Markdown
from rich.prompt import Prompt
from chat_manager import ChatManager
from config import Config

console = Console()

def show_welcome():
    """Display welcome message"""
    welcome_text = """
# 🤖 AI Chat Assistant

Welcome to the OpenAI-powered intelligent chat tool!

## Available Commands:
- `/help` - Show help information
- `/clear` - Clear conversation history
- `/stats` - Display conversation statistics
- `/quit` - Exit program

Start chatting! Type your question...
    """
    console.print(Panel(Markdown(welcome_text), border_style="blue"))

def show_help():
    """Display help information"""
    help_text = """
## 📖 User Guide

### Basic Usage
Simply type your question, and AI will answer. Supports multi-turn conversations where AI remembers context.

### Special Commands
- `/help` - Display this help information
- `/clear` - Clear conversation history, start new conversation
- `/stats` - View conversation statistics (rounds, tokens, etc.)
- `/quit` or `/exit` - Exit program

### Tips
- Clearer questions get more accurate answers
- You can ask AI to respond in specific formats
- Conversation history affects token consumption
    """
    console.print(Panel(Markdown(help_text), border_style="green"))

def show_stats(manager: ChatManager):
    """Display conversation statistics"""
    stats = manager.get_stats()
    stats_text = f"""
## 📊 Conversation Statistics

- Conversation rounds: {stats['rounds']}
- Total messages: {stats['messages']}
- Token consumption: {stats['total_tokens']}
- Model used: {stats['model']}
    """
    console.print(Panel(Markdown(stats_text), border_style="yellow"))

def main():
    """Main program"""
    # Validate configuration
    try:
        Config.validate()
    except ValueError as e:
        console.print(f"[red]{e}[/red]")
        return

    # Display welcome message
    show_welcome()

    # Create chat manager
    manager = ChatManager()

    # Main loop
    while True:
        try:
            # Get user input
            user_input = Prompt.ask("\n[bold cyan]You[/bold cyan]")

            # Handle empty input
            if not user_input.strip():
                continue

            # Handle commands
            if user_input.startswith('/'):
                command = user_input.lower().strip()

                if command == '/quit' or command == '/exit':
                    console.print("[yellow]Goodbye! 👋[/yellow]")
                    break

                elif command == '/help':
                    show_help()
                    continue

                elif command == '/clear':
                    manager.clear_history()
                    console.print("[green]✅ Conversation history cleared[/green]")
                    continue

                elif command == '/stats':
                    show_stats(manager)
                    continue

                else:
                    console.print(f"[red]Unknown command: {command}[/red]")
                    console.print("[yellow]Type /help to see available commands[/yellow]")
                    continue

            # Send chat request
            with console.status("[bold green]AI thinking..."):
                response = manager.chat(user_input)

            # Display response
            if response:
                console.print(Panel(
                    Markdown(response),
                    title="[bold green]🤖 AI Assistant[/bold green]",
                    border_style="green"
                ))
            else:
                console.print("[red]Sorry, unable to get response, please try again[/red]")

        except KeyboardInterrupt:
            console.print("\n[yellow]Interrupt detected, exiting...👋[/yellow]")
            break

        except Exception as e:
            console.print(f"[red]Error: {e}[/red]")

if __name__ == "__main__":
    main()

"""Streaming chat functionality"""
import sys
import time
from openai import OpenAI
from config import Config

class StreamChatManager:
    """Chat manager with streaming output support"""

    def __init__(self, model: str = None):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = model or Config.DEFAULT_MODEL
        self.conversation_history = []

    def chat_stream(self, user_input: str, **kwargs):
        """Streaming chat"""
        self.conversation_history.append({
            "role": "user",
            "content": user_input
        })

        try:
            stream = self.client.chat.completions.create(
                model=self.model,
                messages=self.conversation_history,
                temperature=kwargs.get('temperature', Config.TEMPERATURE),
                max_tokens=kwargs.get('max_tokens', Config.MAX_TOKENS),
                stream=True
            )

            full_response = ""
            print("\n🤖 AI Assistant: ", end="", flush=True)

            for chunk in stream:
                if chunk.choices[0].delta.content:
                    content = chunk.choices[0].delta.content
                    print(content, end="", flush=True)
                    full_response += content
                    time.sleep(0.01)  # Typewriter effect

            print("\n")  # New line

            # Save complete response
            self.conversation_history.append({
                "role": "assistant",
                "content": full_response
            })

            return full_response

        except Exception as e:
            print(f"\n❌ Error: {e}")
            return None

2. Conversation Export Feature

"""Conversation export functionality"""
import os
from datetime import datetime
from typing import List, Dict

class ConversationExporter:
    """Conversation export tool"""

    @staticmethod
    def export_to_markdown(
        conversation: List[Dict],
        filepath: str = None,
        stats: Dict = None
    ) -> str:
        """Export conversation to Markdown format"""

        # Generate filename
        if not filepath:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filepath = f"conversation_{timestamp}.md"

        # Build Markdown content
        lines = []
        lines.append("# AI Conversation Record\n")
        lines.append(f"**Export time:** {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")

        # Add statistics
        if stats:
            lines.append("## 📊 Conversation Statistics\n")
            lines.append(f"- Conversation rounds: {stats.get('rounds', 0)}")
            lines.append(f"- Total messages: {stats.get('messages', 0)}")
            lines.append(f"- Token consumption: {stats.get('total_tokens', 0)}")
            lines.append(f"- Model used: {stats.get('model', 'N/A')}\n")

        lines.append("## 💬 Conversation Content\n")

        # Add conversation
        for i, msg in enumerate(conversation, 1):
            role = msg['role']
            content = msg['content']

            if role == 'user':
                lines.append(f"### 👤 User (#{i//2 + 1})\n")
            else:
                lines.append(f"### 🤖 AI Assistant (#{i//2})\n")

            lines.append(f"{content}\n")

        # Write to file
        content = "\n".join(lines)
        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(content)

        return filepath

# Usage example
def export_conversation(manager):
    """Export conversation"""
    exporter = ConversationExporter()
    filepath = exporter.export_to_markdown(
        conversation=manager.conversation_history,
        stats=manager.get_stats()
    )
    print(f"✅ Conversation exported to: {filepath}")

"""Multi-model management"""
from typing import Dict, List
from rich.console import Console
from rich.table import Table

class ModelManager:
    """Model manager"""

    MODELS = {
        "gpt-3.5-turbo": {
            "name": "GPT-3.5 Turbo",
            "description": "Fast, economical choice",
            "input_price": "$0.0005/1K tokens",
            "output_price": "$0.0015/1K tokens",
            "max_tokens": 4096
        },
        "gpt-4": {
            "name": "GPT-4",
            "description": "Most powerful model",
            "input_price": "$0.03/1K tokens",
            "output_price": "$0.06/1K tokens",
            "max_tokens": 8192
        },
        "gpt-4-turbo-preview": {
            "name": "GPT-4 Turbo",
            "description": "Optimized version of GPT-4",
            "input_price": "$0.01/1K tokens",
            "output_price": "$0.03/1K tokens",
            "max_tokens": 128000
        }
    }

    @classmethod
    def list_models(cls):
        """List all available models"""
        console = Console()
        table = Table(title="🤖 Available Models")

        table.add_column("Model ID", style="cyan")
        table.add_column("Name", style="green")
        table.add_column("Description", style="yellow")
        table.add_column("Input Price", style="magenta")
        table.add_column("Output Price", style="magenta")

        for model_id, info in cls.MODELS.items():
            table.add_row(
                model_id,
                info["name"],
                info["description"],
                info["input_price"],
                info["output_price"]
            )

        console.print(table)

    @classmethod
    def get_model_info(cls, model_id: str) -> Dict:
        """Get model information"""
        return cls.MODELS.get(model_id, None)

    @classmethod
    def validate_model(cls, model_id: str) -> bool:
        """Validate if model is available"""
        return model_id in cls.MODELS

# Add model switching command in CLI
def handle_model_command(manager, args):
    """Handle model-related commands"""
    if not args or args == 'list':
        ModelManager.list_models()
    else:
        model_id = args.strip()
        if ModelManager.validate_model(model_id):
            manager.model = model_id
            info = ModelManager.get_model_info(model_id)
            console.print(f"[green]✅ Switched to model: {info['name']}[/green]")
        else:
            console.print(f"[red]❌ Unknown model: {model_id}[/red]")
            console.print("[yellow]Use /model list to see available models[/yellow]")

"""Enhanced CLI - Integrating all advanced features"""
from rich.console import Console
from rich.panel import Panel
from rich.markdown import Markdown
from rich.prompt import Prompt, Confirm
from stream_chat import StreamChatManager
from exporter import ConversationExporter
from model_manager import ModelManager

console = Console()

def show_advanced_help():
    """Display advanced features help"""
    help_text = """
## 📖 Advanced Features Guide

### Basic Commands
- `/help` - Show help
- `/quit` - Exit program

### Conversation Management
- `/clear` - Clear conversation history
- `/stats` - Display statistics
- `/export` - Export conversation record

### Model Management
- `/model list` - List all models
- `/model ` - Switch model

### Advanced Settings
- `/stream on/off` - Enable/disable streaming output
- `/temperature ` - Set temperature parameter (0-2)
    """
    console.print(Panel(Markdown(help_text), border_style="blue"))

def main():
    """Enhanced main program"""
    # Initialize
    ModelManager.list_models()
    manager = StreamChatManager()
    exporter = ConversationExporter()
    stream_enabled = True

    console.print("[bold green]🚀 Welcome to Enhanced AI Chat Tool![/bold green]")
    console.print("[yellow]Type /help to see all features[/yellow]\n")

    while True:
        try:
            user_input = Prompt.ask("\n[bold cyan]You[/bold cyan]")

            if not user_input.strip():
                continue

            # Handle commands
            if user_input.startswith('/'):
                parts = user_input.split(maxsplit=1)
                command = parts[0].lower()
                args = parts[1] if len(parts) > 1 else ""

                if command in ['/quit', '/exit']:
                    if Confirm.ask("Are you sure you want to quit?"):
                        console.print("[yellow]Goodbye! 👋[/yellow]")
                        break
                    continue

                elif command == '/help':
                    show_advanced_help()
                    continue

                elif command == '/export':
                    filepath = exporter.export_to_markdown(
                        manager.conversation_history,
                        stats=manager.get_stats()
                    )
                    console.print(f"[green]✅ Exported to: {filepath}[/green]")
                    continue

                elif command == '/model':
                    handle_model_command(manager, args)
                    continue

                elif command == '/stream':
                    if args.lower() == 'on':
                        stream_enabled = True
                        console.print("[green]✅ Streaming output enabled[/green]")
                    elif args.lower() == 'off':
                        stream_enabled = False
                        console.print("[yellow]Streaming output disabled[/yellow]")
                    continue

            # Send request
            if stream_enabled:
                manager.chat_stream(user_input)
            else:
                with console.status("[bold green]AI thinking..."):
                    response = manager.chat(user_input)
                if response:
                    console.print(Panel(
                        Markdown(response),
                        title="[bold green]🤖 AI Assistant[/bold green]",
                        border_style="green"
                    ))

        except KeyboardInterrupt:
            console.print("\n[yellow]Exiting...👋[/yellow]")
            break
        except Exception as e:
            console.print(f"[red]Error: {e}[/red]")

if __name__ == "__main__":
    main()

class Config:
    """集中管理配置"""

    # API配置
    OPENAI_API_KEY = os.getenv('OPENAI_API_KEY')
    OPENAI_ORG_ID = os.getenv('OPENAI_ORG_ID')

    # 模型配置
    DEFAULT_MODEL = "gpt-3.5-turbo"
    MAX_TOKENS = 1000
    TEMPERATURE = 0.7

    @classmethod
    def validate(cls):
        """验证配置是否完整"""
        if not cls.OPENAI_API_KEY:
            raise ValueError("❌ OPENAI_API_KEY未设置")
        if not cls.OPENAI_API_KEY.startswith('sk-'):
            raise ValueError("❌ API密钥格式错误")
        print("✅ 配置验证通过")

    @classmethod
    def display(cls):
        """显示配置（隐藏敏感信息）"""
        print("\n📋 当前配置:")
        print(f"  模型: {cls.DEFAULT_MODEL}")
        print(f"  最大Token: {cls.MAX_TOKENS}")
        print(f"  Temperature: {cls.TEMPERATURE}")
        print(f"  API密钥: {'*' * 20}{cls.OPENAI_API_KEY[-8:]}")

# 测试

    Config.validate()
    Config.display()

# detailed_chat.py
import openai
from config import Config
from rich.console import Console
from rich.panel import Panel

console = Console()

def chat_with_gpt(
    prompt: str,
    model: str = "gpt-3.5-turbo",
    temperature: float = 0.7,
    max_tokens: int = 500
) -> dict:
    """
    与GPT对话的函数

    参数:
        prompt: 用户输入
        model: 使用的模型
        temperature: 温度参数(0-2)
        max_tokens: 最大token数

    返回:
        包含响应和元数据的字典
    """

    client = openai.OpenAI(api_key=Config.OPENAI_API_KEY)

    console.print(f"[cyan]🤖 调用模型: {model}[/cyan]")
    console.print(f"[cyan]📊 参数: temp={temperature}, max_tokens={max_tokens}[/cyan]")

    # 调用API
    response = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": "你是一个友好、专业的AI助手。"},
            {"role": "user", "content": prompt}
        ],
        temperature=temperature,
        max_tokens=max_tokens
    )

    # 提取信息
    message = response.choices[0].message.content
    usage = response.usage

    # 计算成本（GPT-3.5-turbo: $0.001/1K tokens）
    cost = (usage.total_tokens / 1000) * 0.001

    # 显示结果
    console.print(Panel(message, title="[green]AI回复[/green]", border_style="green"))

    console.print(f"\n[yellow]📈 使用统计:[/yellow]")
    console.print(f"  输入Token: {usage.prompt_tokens}")
    console.print(f"  输出Token: {usage.completion_tokens}")
    console.print(f"  总Token: {usage.total_tokens}")
    console.print(f"  估算成本: ${cost:.6f}")

    return {
        'message': message,
        'usage': {
            'prompt_tokens': usage.prompt_tokens,
            'completion_tokens': usage.completion_tokens,
            'total_tokens': usage.total_tokens
        },
        'cost': cost,
        'model': model
    }

# 测试

    result = chat_with_gpt(
        "请解释什么是机器学习，用简单的语言",
        temperature=0.7
    )

    print(f"\n✅ 调用成功！")

⚠️ 错误处理与异常

1. 常见错误类型

AuthenticationError

API密钥无效或未设置

RateLimitError

超过速率限制

APIError

OpenAI服务器错误

Timeout

请求超时

2. 完整的错误处理代码

# robust_chat.py
import openai
from openai import OpenAI
import time
from typing import Optional

def chat_with_retry(
    prompt: str,
    max_retries: int = 3,
    retry_delay: int = 2
) -> Optional[dict]:
    """
    带重试机制的对话函数
    """

    client = OpenAI(api_key=Config.OPENAI_API_KEY)

    for attempt in range(max_retries):
        try:
            print(f"📤 尝试 {attempt + 1}/{max_retries}...")

            response = client.chat.completions.create(
                model="gpt-3.5-turbo",
                messages=[{"role": "user", "content": prompt}],
                timeout=30  # 30秒超时
            )

            print("✅ 请求成功！")
            return {
                'message': response.choices[0].message.content,
                'usage': response.usage,
                'success': True
            }

        except openai.AuthenticationError as e:
            print(f"❌ 认证错误: {e}")
            print("💡 请检查API密钥是否正确")
            return {'success': False, 'error': 'authentication_error'}

        except openai.RateLimitError as e:
            print(f"⏳ 速率限制: {e}")
            if attempt < max_retries - 1:
                wait_time = retry_delay * (2 ** attempt)  # 指数退避
                print(f"💤 等待 {wait_time} 秒后重试...")
                time.sleep(wait_time)
            else:
                print("❌ 达到最大重试次数")
                return {'success': False, 'error': 'rate_limit_exceeded'}

        except openai.APIError as e:
            print(f"🔧 API错误: {e}")
            if attempt < max_retries - 1:
                print(f"💤 等待 {retry_delay} 秒后重试...")
                time.sleep(retry_delay)
            else:
                return {'success': False, 'error': 'api_error'}

        except openai.APITimeoutError as e:
            print(f"⏰ 请求超时: {e}")
            if attempt < max_retries - 1:
                print("💤 重试中...")
                time.sleep(retry_delay)
            else:
                return {'success': False, 'error': 'timeout'}

        except Exception as e:
            print(f"❓ 未知错误: {e}")
            return {'success': False, 'error': 'unknown_error'}

    return {'success': False, 'error': 'max_retries_exceeded'}

# 测试
if __name__ == "__main__":
    Config.validate()

    result = chat_with_retry(
        "你好！",
        max_retries=3,
        retry_delay=2
    )

    if result['success']:
        print(f"\n回复: {result['message']}")
    else:
        print(f"\n失败原因: {result['error']}")

💡 错误处理最佳实践

  • ✅ 使用try-except捕获特定异常
  • ✅ 实现重试机制（特别是临时错误）
  • ✅ 使用指数退避策略
  • ✅ 设置合理的超时时间
  • ✅ 记录详细的错误日志
  • ✅ 向用户提供友好的错误提示

# chat_manager.py
from openai import OpenAI
from typing import List, Dict
from config import Config

class ChatManager:
    """对话管理器"""

    def __init__(self, model: str = "gpt-3.5-turbo"):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = model
        self.conversation_history: List[Dict] = []

    def add_system_message(self, content: str):
        """添加系统消息"""
        self.conversation_history.append({
            "role": "system",
            "content": content
        })

    def add_user_message(self, content: str):
        """添加用户消息"""
        self.conversation_history.append({
            "role": "user",
            "content": content
        })

    def add_assistant_message(self, content: str):
        """添加助手消息"""
        self.conversation_history.append({
            "role": "assistant",
            "content": content
        })

    def chat(self, user_input: str, **kwargs) -> str:
        """
        发送消息并获取回复

        参数:
            user_input: 用户输入
            **kwargs: 额外的API参数
        """
        # 添加用户消息到历史
        self.add_user_message(user_input)

        try:
            # 调用API
            response = self.client.chat.completions.create(
                model=self.model,
                messages=self.conversation_history,
                temperature=kwargs.get('temperature', 0.7),
                max_tokens=kwargs.get('max_tokens', 500)
            )

            # 获取回复
            assistant_message = response.choices[0].message.content

            # 添加到历史
            self.add_assistant_message(assistant_message)

            return assistant_message

        except Exception as e:
            print(f"❌ 错误: {e}")
            return "抱歉，出现了错误。"

    def get_history(self) -> List[Dict]:
        """获取对话历史"""
        return self.conversation_history

    def clear_history(self):
        """清空对话历史"""
        self.conversation_history = []

    def save_history(self, filename: str):
        """保存对话历史"""
        import json
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(self.conversation_history, f,
                     ensure_ascii=False, indent=2)

    def load_history(self, filename: str):
        """加载对话历史"""
        import json
        with open(filename, 'r', encoding='utf-8') as f:
            self.conversation_history = json.load(f)

# 使用示例
if __name__ == "__main__":
    Config.validate()

    # 创建对话管理器
    chat = ChatManager()

    # 设置系统提示
    chat.add_system_message("你是一个友好的AI助手，擅长解答问题。")

    # 对话
    response1 = chat.chat("你好！")
    print(f"助手: {response1}\n")

    response2 = chat.chat("你能帮我什么？")
    print(f"助手: {response2}\n")

    # 保存历史
    chat.save_history("conversation.json")

# cli_chat.py
from chat_manager import ChatManager
from config import Config
from rich.console import Console
from rich.panel import Panel
from rich.markdown import Markdown
from rich.prompt import Prompt
import sys

console = Console()

def print_welcome():
    """打印欢迎信息"""
    welcome_text = """
# 🤖 AI聊天助手

欢迎使用AI命令行聊天工具！

**可用命令:**
- `/help` - 显示帮助
- `/clear` - 清空对话历史
- `/save` - 保存对话
- `/history` - 查看历史
- `/quit` - 退出程序

直接输入消息开始聊天吧！
    """
    console.print(Panel(Markdown(welcome_text),
                       border_style="blue",
                       title="欢迎"))

def main():
    """主函数"""
    Config.validate()

    # 打印欢迎信息
    print_welcome()

    # 创建对话管理器
    chat = ChatManager()
    chat.add_system_message("你是一个友好、有帮助的AI助手。")

    # 主循环
    while True:
        try:
            # 获取用户输入
            user_input = Prompt.ask("\n[bold cyan]你[/bold cyan]")

            # 处理命令
            if user_input.startswith('/'):
                handle_command(user_input, chat)
                continue

            # 如果为空，继续
            if not user_input.strip():
                continue

            # 显示思考状态
            with console.status("[bold green]AI正在思考中...",
                              spinner="dots"):
                response = chat.chat(user_input)

            # 显示回复
            console.print(Panel(
                Markdown(response),
                title="[bold green]🤖 AI助手[/bold green]",
                border_style="green"
            ))

        except KeyboardInterrupt:
            console.print("\n\n[yellow]检测到 Ctrl+C，正在退出...[/yellow]")
            save_before_exit(chat)
            break

        except Exception as e:
            console.print(f"[red]❌ 错误: {e}[/red]")

def handle_command(command: str, chat: ChatManager):
    """处理命令"""
    command = command.lower().strip()

    if command == '/help':
        print_welcome()

    elif command == '/clear':
        chat.clear_history()
        console.print("[green]✅ 对话历史已清空[/green]")

    elif command == '/save':
        filename = Prompt.ask("保存文件名", default="conversation.json")
        chat.save_history(filename)
        console.print(f"[green]✅ 已保存到 {filename}[/green]")

    elif command == '/history':
        history = chat.get_history()
        if not history:
            console.print("[yellow]没有对话历史[/yellow]")
        else:
            for i, msg in enumerate(history, 1):
                role = msg['role']
                content = msg['content']
                color = "cyan" if role == "user" else "green"
                console.print(f"[{color}]{i}. {role}: {content}[/{color}]")

    elif command == '/quit':
        save_before_exit(chat)
        sys.exit(0)

    else:
        console.print(f"[red]未知命令: {command}[/red]")
        console.print("[yellow]输入 /help 查看帮助[/yellow]")

def save_before_exit(chat: ChatManager):
    """退出前保存"""
    if chat.get_history():
        save = Prompt.ask("是否保存对话历史？",
                         choices=["y", "n"],
                         default="y")
        if save == "y":
            filename = Prompt.ask("文件名",
                                 default="conversation.json")
            chat.save_history(filename)
            console.print(f"[green]✅ 已保存[/green]")

    console.print("[blue]👋 再见！[/blue]")

if __name__ == "__main__":
    main()

# streaming_chat.py
def chat_stream(prompt: str):
    """流式输出对话"""
    client = OpenAI(api_key=Config.OPENAI_API_KEY)

    print("🤖 AI助手: ", end="", flush=True)

    stream = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}],
        stream=True  # 启用流式输出
    )

    full_response = ""
    for chunk in stream:
        if chunk.choices[0].delta.content is not None:
            content = chunk.choices[0].delta.content
            print(content, end="", flush=True)
            full_response += content

    print()  # 换行
    return full_response

# 使用
if __name__ == "__main__":
    Config.validate()
    chat_stream("请讲一个简短的故事")

# multi_model_chat.py
class MultiModelChat:
    """支持多个模型的聊天类"""

    MODELS = {
        '1': {'name': 'gpt-3.5-turbo', 'desc': '快速、经济'},
        '2': {'name': 'gpt-4o-mini', 'desc': '平衡性能'},
        '3': {'name': 'gpt-4o', 'desc': '最强性能'},
    }

    def __init__(self):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.current_model = 'gpt-3.5-turbo'

    def select_model(self):
        """选择模型"""
        console.print("\n[bold]可用模型:[/bold]")
        for key, info in self.MODELS.items():
            console.print(f"{key}. {info['name']} - {info['desc']}")

        choice = Prompt.ask("选择模型",
                           choices=list(self.MODELS.keys()),
                           default='1')
        self.current_model = self.MODELS[choice]['name']
        console.print(f"[green]✅ 已切换到 {self.current_model}[/green]")

    def chat(self, prompt: str) -> str:
        response = self.client.chat.completions.create(
            model=self.current_model,
            messages=[{"role": "user", "content": prompt}]
        )
        return response.choices[0].message.content

# export_utils.py
import json
from datetime import datetime

def export_to_markdown(history: list, filename: str):
    """导出为Markdown格式"""
    with open(filename, 'w', encoding='utf-8') as f:
        f.write(f"# AI对话记录\n\n")
        f.write(f"**导出时间**: {datetime.now()}\n\n")
        f.write("---\n\n")

        for msg in history:
            role = msg['role']
            content = msg['content']

            if role == "user":
                f.write(f"### 👤 用户\n{content}\n\n")
            elif role == "assistant":
                f.write(f"### 🤖 AI助手\n{content}\n\n")

    print(f"✅ 已导出到 {filename}")

def export_to_html(history: list, filename: str):
    """导出为HTML格式"""
    html = """
    
    
    
        
        AI对话记录
        
    
    
        
AI对话记录

        
导出时间: {}

    """.format(datetime.now())

    for msg in history:
        role = msg['role']
        content = msg['content']
        css_class = 'user' if role == 'user' else 'assistant'
        icon = '👤' if role == 'user' else '🤖'
        html += f'
{icon} {role}
{content}
\n'

    html += ""

    with open(filename, 'w', encoding='utf-8') as f:
        f.write(html)

    print(f"✅ 已导出到 {filename}")

import os

class Config:
    """Centralized configuration management"""

    # API configuration
    OPENAI_API_KEY = os.getenv('OPENAI_API_KEY')
    OPENAI_ORG_ID = os.getenv('OPENAI_ORG_ID')

    # Model configuration
    DEFAULT_MODEL = "gpt-3.5-turbo"
    MAX_TOKENS = 1000
    TEMPERATURE = 0.7

    @classmethod
    def validate(cls):
        """Validate configuration completeness"""
        if not cls.OPENAI_API_KEY:
            raise ValueError("❌ OPENAI_API_KEY not set")
        if not cls.OPENAI_API_KEY.startswith('sk-'):
            raise ValueError("❌ Invalid API key format")
        print("✅ Configuration validated")

    @classmethod
    def display(cls):
        """Display configuration (hide sensitive info)"""
        print("\n📋 Current Configuration:")
        print(f"  Model: {cls.DEFAULT_MODEL}")
        print(f"  Max Tokens: {cls.MAX_TOKENS}")
        print(f"  Temperature: {cls.TEMPERATURE}")
        print(f"  API Key: {'*' * 20}{cls.OPENAI_API_KEY[-8:]}")

# Test
if __name__ == "__main__":
    Config.validate()
    Config.display()

# detailed_chat.py
import openai
from config import Config
from rich.console import Console
from rich.panel import Panel

console = Console()

def chat_with_gpt(
    prompt: str,
    model: str = "gpt-3.5-turbo",
    temperature: float = 0.7,
    max_tokens: int = 500
) -> dict:
    """
    Chat with GPT function

    Args:
        prompt: User input
        model: Model to use
        temperature: Temperature parameter (0-2)
        max_tokens: Maximum tokens

    Returns:
        Dictionary containing response and metadata
    """

    client = openai.OpenAI(api_key=Config.OPENAI_API_KEY)

    console.print(f"[cyan]🤖 Calling model: {model}[/cyan]")
    console.print(f"[cyan]📊 Parameters: temp={temperature}, max_tokens={max_tokens}[/cyan]")

    # Call API
    response = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": "You are a friendly, professional AI assistant."},
            {"role": "user", "content": prompt}
        ],
        temperature=temperature,
        max_tokens=max_tokens
    )

    # Extract information
    message = response.choices[0].message.content
    usage = response.usage

    # Calculate cost (GPT-3.5-turbo: $0.001/1K tokens)
    cost = (usage.total_tokens / 1000) * 0.001

    # Display results
    console.print(Panel(message, title="[green]AI Response[/green]", border_style="green"))

    console.print(f"\n[yellow]📈 Usage Statistics:[/yellow]")
    console.print(f"  Input Tokens: {usage.prompt_tokens}")
    console.print(f"  Output Tokens: {usage.completion_tokens}")
    console.print(f"  Total Tokens: {usage.total_tokens}")
    console.print(f"  Estimated Cost: ${cost:.6f}")

    return {
        'message': message,
        'usage': {
            'prompt_tokens': usage.prompt_tokens,
            'completion_tokens': usage.completion_tokens,
            'total_tokens': usage.total_tokens
        },
        'cost': cost,
        'model': model
    }

    result = chat_with_gpt(
        "Please explain what machine learning is in simple terms",
        temperature=0.7
    )

# robust_chat.py
import openai
from openai import OpenAI
import time
from typing import Optional

def chat_with_retry(
    prompt: str,
    max_retries: int = 3,
    retry_delay: int = 2
) -> Optional[dict]:
    """
    Chat function with retry mechanism
    """

    client = OpenAI(api_key=Config.OPENAI_API_KEY)

    for attempt in range(max_retries):
        try:
            print(f"📤 Attempt {attempt + 1}/{max_retries}...")

            response = client.chat.completions.create(
                model="gpt-3.5-turbo",
                messages=[{"role": "user", "content": prompt}],
                timeout=30  # 30 second timeout
            )

            print("✅ Request successful!")
            return {
                'message': response.choices[0].message.content,
                'usage': response.usage,
                'success': True
            }

        except openai.AuthenticationError as e:
            print(f"❌ Authentication error: {e}")
            print("💡 Please check if API key is correct")
            return {'success': False, 'error': 'authentication_error'}

        except openai.RateLimitError as e:
            print(f"⏳ Rate limit: {e}")
            if attempt < max_retries - 1:
                wait_time = retry_delay * (2 ** attempt)  # Exponential backoff
                print(f"💤 Waiting {wait_time} seconds before retry...")
                time.sleep(wait_time)
            else:
                print("❌ Max retries reached")
                return {'success': False, 'error': 'rate_limit_exceeded'}

        except openai.APIError as e:
            print(f"🔧 API error: {e}")
            if attempt < max_retries - 1:
                print(f"💤 Waiting {retry_delay} seconds before retry...")
                time.sleep(retry_delay)
            else:
                return {'success': False, 'error': 'api_error'}

        except openai.APITimeoutError as e:
            print(f"⏰ Request timeout: {e}")
            if attempt < max_retries - 1:
                print("💤 Retrying...")
                time.sleep(retry_delay)
            else:
                return {'success': False, 'error': 'timeout'}

        except Exception as e:
            print(f"❓ Unknown error: {e}")
            return {'success': False, 'error': 'unknown_error'}

    return {'success': False, 'error': 'max_retries_exceeded'}

# Test
if __name__ == "__main__":
    Config.validate()

    result = chat_with_retry(
        "Hello!",
        max_retries=3,
        retry_delay=2
    )

    if result['success']:
        print(f"\nResponse: {result['message']}")
    else:
        print(f"\nFailed: {result['error']}")

# chat_manager.py
from openai import OpenAI
from typing import List, Dict
from config import Config

class ChatManager:
    """Chat Manager"""

    def __init__(self, model: str = "gpt-3.5-turbo"):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.model = model
        self.conversation_history: List[Dict] = []

    def add_system_message(self, content: str):
        """Add system message"""
        self.conversation_history.append({
            "role": "system",
            "content": content
        })

    def add_user_message(self, content: str):
        """Add user message"""
        self.conversation_history.append({
            "role": "user",
            "content": content
        })

    def add_assistant_message(self, content: str):
        """Add assistant message"""
        self.conversation_history.append({
            "role": "assistant",
            "content": content
        })

    def chat(self, user_input: str, **kwargs) -> str:
        """
        Send message and get response

        Args:
            user_input: User input
            **kwargs: Additional API parameters
        """
        # Add user message to history
        self.add_user_message(user_input)

        try:
            # Call API
            response = self.client.chat.completions.create(
                model=self.model,
                messages=self.conversation_history,
                temperature=kwargs.get('temperature', 0.7),
                max_tokens=kwargs.get('max_tokens', 500)
            )

            # Get response
            assistant_message = response.choices[0].message.content

            # Add to history
            self.add_assistant_message(assistant_message)

            return assistant_message

        except Exception as e:
            print(f"❌ Error: {e}")
            return "Sorry, an error occurred."

    def get_history(self) -> List[Dict]:
        """Get conversation history"""
        return self.conversation_history

    def clear_history(self):
        """Clear conversation history"""
        self.conversation_history = []

    def save_history(self, filename: str):
        """Save conversation history"""
        import json
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(self.conversation_history, f,
                     ensure_ascii=False, indent=2)

    def load_history(self, filename: str):
        """Load conversation history"""
        import json
        with open(filename, 'r', encoding='utf-8') as f:
            self.conversation_history = json.load(f)

# cli_chat.py
from chat_manager import ChatManager
from config import Config
from rich.console import Console
from rich.panel import Panel
from rich.markdown import Markdown
from rich.prompt import Prompt
import sys

console = Console()

def print_welcome():
    """Print welcome message"""
    welcome_text = """
# 🤖 AI Chat Assistant

Welcome to the AI command-line chat tool!

**Available Commands:**
- `/help` - Show help
- `/clear` - Clear conversation history
- `/save` - Save conversation
- `/history` - View history
- `/quit` - Exit program

Start chatting by typing a message!
    """
    console.print(Panel(Markdown(welcome_text),
                       border_style="blue",
                       title="Welcome"))

def main():
    """Main function"""
    Config.validate()

    # Print welcome message
    print_welcome()

    # Create chat manager
    chat = ChatManager()
    chat.add_system_message("You are a friendly, helpful AI assistant.")

    # Main loop
    while True:
        try:
            # Get user input
            user_input = Prompt.ask("\n[bold cyan]You[/bold cyan]")

            # Handle commands
            if user_input.startswith('/'):
                handle_command(user_input, chat)
                continue

            # If empty, continue
            if not user_input.strip():
                continue

            # Show thinking status
            with console.status("[bold green]AI is thinking...",
                              spinner="dots"):
                response = chat.chat(user_input)

            # Display response
            console.print(Panel(
                Markdown(response),
                title="[bold green]🤖 AI Assistant[/bold green]",
                border_style="green"
            ))

        except KeyboardInterrupt:
            console.print("\n\n[yellow]Ctrl+C detected, exiting...[/yellow]")
            save_before_exit(chat)
            break

        except Exception as e:
            console.print(f"[red]❌ Error: {e}[/red]")

def handle_command(command: str, chat: ChatManager):
    """Handle commands"""
    command = command.lower().strip()

    if command == '/help':
        print_welcome()

    elif command == '/clear':
        chat.clear_history()
        console.print("[green]✅ Conversation history cleared[/green]")

    elif command == '/save':
        filename = Prompt.ask("Save filename", default="conversation.json")
        chat.save_history(filename)
        console.print(f"[green]✅ Saved to {filename}[/green]")

    elif command == '/history':
        history = chat.get_history()
        if not history:
            console.print("[yellow]No conversation history[/yellow]")
        else:
            for i, msg in enumerate(history, 1):
                role = msg['role']
                content = msg['content']
                color = "cyan" if role == "user" else "green"
                console.print(f"[{color}]{i}. {role}: {content}[/{color}]")

    elif command == '/quit':
        save_before_exit(chat)
        sys.exit(0)

    else:
        console.print(f"[red]Unknown command: {command}[/red]")
        console.print("[yellow]Type /help for help[/yellow]")

def save_before_exit(chat: ChatManager):
    """Save before exit"""
    if chat.get_history():
        save = Prompt.ask("Save conversation history?",
                         choices=["y", "n"],
                         default="y")
        if save == "y":
            filename = Prompt.ask("Filename",
                                 default="conversation.json")
            chat.save_history(filename)
            console.print(f"[green]✅ Saved[/green]")

    console.print("[blue]👋 Goodbye![/blue]")

if __name__ == "__main__":
    main()

# streaming_chat.py
def chat_stream(prompt: str):
    """Stream output chat"""
    client = OpenAI(api_key=Config.OPENAI_API_KEY)

    print("🤖 AI Assistant: ", end="", flush=True)

    stream = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}],
        stream=True  # Enable streaming
    )

    full_response = ""
    for chunk in stream:
        if chunk.choices[0].delta.content is not None:
            content = chunk.choices[0].delta.content
            print(content, end="", flush=True)
            full_response += content

    print()  # New line
    return full_response

# Usage
if __name__ == "__main__":
    Config.validate()
    chat_stream("Please tell a short story")

# multi_model_chat.py
class MultiModelChat:
    """Chat class supporting multiple models"""

    MODELS = {
        '1': {'name': 'gpt-3.5-turbo', 'desc': 'Fast, economical'},
        '2': {'name': 'gpt-4o-mini', 'desc': 'Balanced performance'},
        '3': {'name': 'gpt-4o', 'desc': 'Best performance'},
    }

    def __init__(self):
        self.client = OpenAI(api_key=Config.OPENAI_API_KEY)
        self.current_model = 'gpt-3.5-turbo'

    def select_model(self):
        """Select model"""
        console.print("\n[bold]Available Models:[/bold]")
        for key, info in self.MODELS.items():
            console.print(f"{key}. {info['name']} - {info['desc']}")

        choice = Prompt.ask("Select model",
                           choices=list(self.MODELS.keys()),
                           default='1')
        self.current_model = self.MODELS[choice]['name']
        console.print(f"[green]✅ Switched to {self.current_model}[/green]")

    def chat(self, prompt: str) -> str:
        response = self.client.chat.completions.create(
            model=self.current_model,
            messages=[{"role": "user", "content": prompt}]
        )
        return response.choices[0].message.content

# export_utils.py
import json
from datetime import datetime

def export_to_markdown(history: list, filename: str):
    """Export as Markdown format"""
    with open(filename, 'w', encoding='utf-8') as f:
        f.write(f"# AI Conversation Log\n\n")
        f.write(f"**Export Time**: {datetime.now()}\n\n")
        f.write("---\n\n")

        for msg in history:
            role = msg['role']
            content = msg['content']

            if role == "user":
                f.write(f"### 👤 User\n{content}\n\n")
            elif role == "assistant":
                f.write(f"### 🤖 AI Assistant\n{content}\n\n")

    print(f"✅ Exported to {filename}")

def export_to_html(history: list, filename: str):
    """Export as HTML format"""
    html = """
    
    
    
        
        AI Conversation Log
        
    
    
        
AI Conversation Log

        
Export Time: {}

    """.format(datetime.now())

    for msg in history:
        role = msg['role']
        content = msg['content']
        css_class = 'user' if role == 'user' else 'assistant'
        icon = '👤' if role == 'user' else '🤖'
        html += f'
{icon} {role}
{content}
\n'

    html += ""

    with open(filename, 'w', encoding='utf-8') as f:
        f.write(html)

    print(f"✅ Exported to {filename}")

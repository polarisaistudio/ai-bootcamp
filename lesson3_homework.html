任务要求

项目描述

开发一个智能学习助手，能够根据用户的学习水平（初级、中级、高级）提供个性化的解释，并通过记忆系统跟踪学习进度，动态调整解释难度。

功能要求（15分）：正确使用Prompt Templates

  • 为不同难度级别创建Prompt模板（beginner、intermediate、advanced）
  • 支持变量插值（主题、用户名、难度级别等）
  • 模板应包含清晰的角色定义和输出格式要求
  • 使用LangChain的PromptTemplate或ChatPromptTemplate

功能要求（15分）：实现Memory系统

  • 使用ConversationBufferMemory或ConversationSummaryMemory
  • 跟踪用户学习的主题历史
  • 记录用户的理解程度反馈
  • 能够清除和查看记忆内容

功能要求（10分）：动态难度调整

  • 根据用户反馈（"太简单"、"刚好"、"太难"）调整难度
  • 实现难度级别的自动升降机制
  • 显示当前难度级别和学习统计

💡 技术要点

  • 使用免费的Ollama模型（llama2、mistral等）
  • Prompt模板应该清晰区分不同难度级别的解释风格
  • Memory应该能够持久化保存（可选：保存到JSON文件）

2. 学习助手主程序 (learning_assistant.py)

"""个性化学习助手 - 作业1"""
from langchain.prompts import ChatPromptTemplate
from langchain.memory import ConversationBufferMemory
from langchain_community.llms import Ollama
from langchain.chains import LLMChain
import json
from datetime import datetime

class LearningAssistant:
    """个性化学习助手"""

    def __init__(self, model_name="llama2"):
        """初始化助手"""
        self.llm = Ollama(model=model_name)
        self.memory = ConversationBufferMemory(
            memory_key="chat_history",
            return_messages=True
        )

        # 难度级别配置
        self.difficulty_levels = {
            "beginner": {
                "name": "初级",
                "description": "使用简单语言，多举例子，避免专业术语"
            },
            "intermediate": {
                "name": "中级",
                "description": "使用标准术语，提供适度深度的解释"
            },
            "advanced": {
                "name": "高级",
                "description": "使用专业术语，深入技术细节和原理"
            }
        }

        self.current_difficulty = "intermediate"
        self.learning_stats = {
            "topics_learned": [],
            "difficulty_changes": [],
            "total_interactions": 0
        }

        # 创建不同难度的Prompt模板
        self.prompt_templates = self._create_prompt_templates()

    def _create_prompt_templates(self):
        """创建不同难度级别的Prompt模板"""
        templates = {}

        # 初级模板
        templates["beginner"] = ChatPromptTemplate.from_template("""
你是一位耐心的导师，正在教一个刚入门的学生。

学生问题：{topic}

请用以下方式解释：
1. 使用简单、日常的语言
2. 避免使用专业术语，如果必须使用，请详细解释
3. 举2-3个生活中的具体例子
4. 用类比的方式帮助理解
5. 回答要简洁，不超过200字

解释：
""")

        # 中级模板
        templates["intermediate"] = ChatPromptTemplate.from_template("""
你是一位专业的技术讲师，正在教一个有一定基础的学生。

学生问题：{topic}

请用以下方式解释：
1. 使用标准的技术术语
2. 提供清晰的概念定义
3. 说明原理和工作机制
4. 举1-2个实际应用场景
5. 回答要全面但不过度深入

解释：
""")

        # 高级模板
        templates["advanced"] = ChatPromptTemplate.from_template("""
你是一位资深技术专家，正在与一个有经验的开发者交流。

技术问题：{topic}

请用以下方式解释：
1. 使用专业术语和行业标准
2. 深入分析技术原理和底层实现
3. 讨论性能、优缺点和最佳实践
4. 提供代码示例或技术细节
5. 可以涉及相关的高级话题

详细解释：
""")

        return templates

    def explain_topic(self, topic: str) -> str:
        """根据当前难度级别解释主题"""
        # 获取当前难度的模板
        template = self.prompt_templates[self.current_difficulty]

        # 创建Chain
        chain = LLMChain(
            llm=self.llm,
            prompt=template,
            memory=self.memory
        )

        # 生成解释
        response = chain.run(topic=topic)

        # 更新统计
        self.learning_stats["topics_learned"].append({
            "topic": topic,
            "difficulty": self.current_difficulty,
            "timestamp": datetime.now().isoformat()
        })
        self.learning_stats["total_interactions"] += 1

        return response

    def adjust_difficulty(self, feedback: str):
        """根据用户反馈调整难度"""
        feedback = feedback.lower()

        old_difficulty = self.current_difficulty

        if "太简单" in feedback or "easy" in feedback:
            if self.current_difficulty == "beginner":
                self.current_difficulty = "intermediate"
            elif self.current_difficulty == "intermediate":
                self.current_difficulty = "advanced"

        elif "太难" in feedback or "hard" in feedback:
            if self.current_difficulty == "advanced":
                self.current_difficulty = "intermediate"
            elif self.current_difficulty == "intermediate":
                self.current_difficulty = "beginner"

        if old_difficulty != self.current_difficulty:
            self.learning_stats["difficulty_changes"].append({
                "from": old_difficulty,
                "to": self.current_difficulty,
                "timestamp": datetime.now().isoformat()
            })
            return f"难度已从 {self.difficulty_levels[old_difficulty]['name']} 调整为 {self.difficulty_levels[self.current_difficulty]['name']}"

        return "难度保持不变"

    def get_stats(self) -> dict:
        """获取学习统计"""
        return {
            "current_difficulty": self.difficulty_levels[self.current_difficulty]['name'],
            "topics_count": len(self.learning_stats["topics_learned"]),
            "total_interactions": self.learning_stats["total_interactions"],
            "difficulty_changes": len(self.learning_stats["difficulty_changes"])
        }

    def save_progress(self, filepath="learning_progress.json"):
        """保存学习进度"""
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(self.learning_stats, f, ensure_ascii=False, indent=2)
        print(f"✅ 学习进度已保存到 {filepath}")

    def clear_memory(self):
        """清除对话记忆"""
        self.memory.clear()
        print("✅ 对话记忆已清除")


def main():
    """主程序"""
    print("🎓 欢迎使用个性化学习助手！\n")

    assistant = LearningAssistant()

    print("可用命令：")
    print("  - 直接输入问题进行学习")
    print("  - /difficulty [太简单/刚好/太难] - 调整难度")
    print("  - /stats - 查看学习统计")
    print("  - /save - 保存学习进度")
    print("  - /clear - 清除对话记忆")
    print("  - /quit - 退出\n")

    while True:
        user_input = input("\n💭 你的问题: ").strip()

        if not user_input:
            continue

        if user_input == "/quit":
            print("\n👋 再见！继续保持学习的热情！")
            break

        elif user_input == "/stats":
            stats = assistant.get_stats()
            print("\n📊 学习统计：")
            print(f"  当前难度：{stats['current_difficulty']}")
            print(f"  学习主题数：{stats['topics_count']}")
            print(f"  总互动次数：{stats['total_interactions']}")
            print(f"  难度调整次数：{stats['difficulty_changes']}")

        elif user_input == "/save":
            assistant.save_progress()

        elif user_input == "/clear":
            assistant.clear_memory()

        elif user_input.startswith("/difficulty"):
            parts = user_input.split(maxsplit=1)
            if len(parts) > 1:
                result = assistant.adjust_difficulty(parts[1])
                print(f"\n{result}")
            else:
                print("\n请提供反馈：/difficulty [太简单/刚好/太难]")

        else:
            # 解释主题
            print(f"\n🎯 当前难度：{assistant.difficulty_levels[assistant.current_difficulty]['name']}")
            print("\n📖 解释：")
            response = assistant.explain_topic(user_input)
            print(response)
            print("\n💡 提示：如果解释太简单或太难，可以使用 /difficulty 调整")


if __name__ == "__main__":
    main()

1. 内容创作系统 (content_creator.py)

"""内容创作流程系统 - 作业2"""
from langchain.prompts import ChatPromptTemplate
from langchain_community.llms import Ollama
from langchain.chains import LLMChain, SequentialChain
from datetime import datetime
import os

class ContentCreator:
    """自动化内容创作系统"""

    def __init__(self, model_name="llama2"):
        """初始化创作系统"""
        self.llm = Ollama(model=model_name)

        # 文章类型配置
        self.article_types = {
            "technical": "技术文章",
            "story": "故事/小说",
            "marketing": "营销文案"
        }

        # 创建输出目录
        self.output_dir = "generated_content"
        os.makedirs(self.output_dir, exist_ok=True)

    def create_title_chain(self, article_type: str):
        """创建标题生成Chain"""
        type_prompts = {
            "technical": "生成一个吸引人的技术文章标题，要求：\n1. 突出技术要点\n2. 包含关键词\n3. 简洁有力（10-15字）",
            "story": "生成一个引人入胜的故事标题，要求：\n1. 激发好奇心\n2. 富有文学性\n3. 简洁优雅（8-12字）",
            "marketing": "生成一个吸引眼球的营销标题，要求：\n1. 突出卖点\n2. 制造紧迫感\n3. 引发行动（10-15字）"
        }

        prompt = ChatPromptTemplate.from_template(f"""
{type_prompts[article_type]}

主题：{{topic}}

只返回标题，不要其他内容。
标题：
""")

        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            output_key="title"
        )

    def create_outline_chain(self, article_type: str):
        """创建大纲生成Chain"""
        type_prompts = {
            "technical": "为技术文章创建详细大纲，包括：\n1. 引言\n2. 技术背景\n3. 核心内容（3-5个要点）\n4. 代码示例说明\n5. 总结",
            "story": "为故事创建情节大纲，包括：\n1. 开端\n2. 发展\n3. 高潮\n4. 结局",
            "marketing": "为营销文案创建结构，包括：\n1. 痛点描述\n2. 解决方案\n3. 产品优势\n4. 行动召唤"
        }

        prompt = ChatPromptTemplate.from_template(f"""
{type_prompts[article_type]}

主题：{{topic}}
标题：{{title}}

请生成结构化的大纲（使用markdown格式）：
""")

        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            output_key="outline"
        )

    def create_content_chain(self, article_type: str):
        """创建正文生成Chain"""
        type_prompts = {
            "technical": "根据大纲撰写技术文章，要求：\n1. 语言专业准确\n2. 包含代码示例\n3. 逻辑清晰\n4. 800-1000字",
            "story": "根据大纲创作故事，要求：\n1. 情节生动\n2. 人物鲜明\n3. 语言优美\n4. 800-1000字",
            "marketing": "根据大纲撰写营销文案，要求：\n1. 语言有说服力\n2. 突出价值\n3. 引导行动\n4. 600-800字"
        }

        prompt = ChatPromptTemplate.from_template(f"""
{type_prompts[article_type]}

主题：{{topic}}
标题：{{title}}
大纲：{{outline}}

请撰写完整的正文内容（使用markdown格式）：
""")

        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            output_key="content"
        )

    def create_article(self, topic: str, article_type: str = "technical"):
        """创建完整文章"""
        if article_type not in self.article_types:
            raise ValueError(f"不支持的文章类型：{article_type}")

        print(f"\n📝 开始创作{self.article_types[article_type]}...")
        print(f"主题：{topic}\n")

        # 创建三个Chain
        title_chain = self.create_title_chain(article_type)
        outline_chain = self.create_outline_chain(article_type)
        content_chain = self.create_content_chain(article_type)

        # 组合成Sequential Chain
        overall_chain = SequentialChain(
            chains=[title_chain, outline_chain, content_chain],
            input_variables=["topic"],
            output_variables=["title", "outline", "content"],
            verbose=True
        )

        # 执行创作流程
        print("🎯 步骤1：生成标题...")
        result = overall_chain({"topic": topic})

        # 显示结果
        print(f"\n✅ 标题：{result['title']}\n")
        print("📋 大纲：")
        print(result['outline'])
        print("\n" + "="*50 + "\n")

        # 保存到文件
        filepath = self.save_article(
            topic=topic,
            title=result['title'],
            outline=result['outline'],
            content=result['content'],
            article_type=article_type
        )

        print(f"\n✅ 文章已保存到：{filepath}")

        return result

    def save_article(self, topic, title, outline, content, article_type):
        """保存文章到文件"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"{article_type}_{timestamp}.md"
        filepath = os.path.join(self.output_dir, filename)

        markdown_content = f"""# {title}

> 主题：{topic}
> 类型：{self.article_types[article_type]}
> 生成时间：{datetime.now().strftime("%Y-%m-%d %H:%M:%S")}

---

## 📋 文章大纲

{outline}

---

## 📖 正文内容

{content}

---

*由AI自动生成*
"""

        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(markdown_content)

        return filepath


def main():
    """主程序"""
    print("✍️ 欢迎使用AI内容创作系统！\n")

    creator = ContentCreator()

    print("支持的文章类型：")
    for key, value in creator.article_types.items():
        print(f"  {key}: {value}")
    print()

    while True:
        topic = input("请输入文章主题（或输入 'quit' 退出）: ").strip()

        if topic.lower() == 'quit':
            print("\n👋 再见！")
            break

        if not topic:
            continue

        article_type = input("请选择文章类型 [technical/story/marketing]: ").strip() or "technical"

        try:
            creator.create_article(topic, article_type)
        except Exception as e:
            print(f"\n❌ 创作失败：{e}")

        print("\n" + "="*60 + "\n")


if __name__ == "__main__":
    main()

1. 智能客服系统 (customer_service.py)

"""智能客服系统 - 作业3"""
from langchain.prompts import ChatPromptTemplate, FewShotChatMessagePromptTemplate
from langchain.memory import ConversationBufferMemory
from langchain_community.llms import Ollama
from langchain.chains import LLMChain
import json
from datetime import datetime
import re

class CustomerServiceBot:
    """智能客服机器人"""

    def __init__(self, model_name="llama2"):
        """初始化客服机器人"""
        self.llm = Ollama(model=model_name)
        self.memory = ConversationBufferMemory(
            memory_key="chat_history",
            return_messages=True
        )

        # 用户信息
        self.user_info = {
            "name": None,
            "questions_asked": [],
            "sentiment_history": []
        }

        # 对话日志
        self.conversation_log = []

        # FAQ示例（Few-shot）
        self.faq_examples = [
            {
                "question": "如何重置密码？",
                "answer": "您可以通过以下步骤重置密码：\n1. 点击登录页面的"忘记密码"\n2. 输入注册邮箱\n3. 查收重置邮件\n4. 点击链接设置新密码"
            },
            {
                "question": "退货政策是什么？",
                "answer": "我们的退货政策如下：\n- 30天内可无理由退货\n- 商品需保持原包装\n- 非质量问题退货需承担运费\n- 填写退货申请单后3-5个工作日处理"
            },
            {
                "question": "支持哪些支付方式？",
                "answer": "我们支持多种支付方式：\n- 信用卡/借记卡\n- 支付宝\n- 微信支付\n- PayPal\n- 银行转账"
            },
            {
                "question": "配送需要多长时间？",
                "answer": "配送时效如下：\n- 同城：1-2个工作日\n- 省内：2-3个工作日\n- 跨省：3-5个工作日\n- 偏远地区：5-7个工作日"
            },
            {
                "question": "如何联系人工客服？",
                "answer": "您可以通过以下方式联系人工客服：\n- 客服热线：400-123-4567（9:00-18:00）\n- 在线客服：点击右下角图标\n- 邮箱：support@company.com"
            }
        ]

        # 创建Few-shot Prompt
        self.faq_chain = self._create_faq_chain()
        self.chat_chain = self._create_chat_chain()

    def _create_faq_chain(self):
        """创建FAQ问答Chain（Few-shot）"""
        # Few-shot示例格式
        example_prompt = ChatPromptTemplate.from_messages([
            ("human", "{question}"),
            ("ai", "{answer}")
        ])

        few_shot_prompt = FewShotChatMessagePromptTemplate(
            example_prompt=example_prompt,
            examples=self.faq_examples
        )

        # 完整Prompt
        final_prompt = ChatPromptTemplate.from_messages([
            ("system", "你是一个专业的客服助手。请参考以下FAQ示例来回答用户问题。"),
            few_shot_prompt,
            ("human", "{input}")
        ])

        return LLMChain(llm=self.llm, prompt=final_prompt)

    def _create_chat_chain(self):
        """创建对话Chain"""
        template = """你是一个友好、专业的客服代表。

用户情绪：{sentiment}
用户姓名：{user_name}

请根据用户情绪调整回复风格：
- 消极情绪：表达同理心，安抚情绪，积极解决问题
- 中性情绪：保持专业、友好
- 积极情绪：保持热情、感谢用户

对话历史：{chat_history}

用户问题：{input}

回复："""

        prompt = ChatPromptTemplate.from_template(template)
        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            memory=self.memory
        )

    def analyze_sentiment(self, text: str) -> str:
        """简单的情绪分析（基于关键词）"""
        text = text.lower()

        # 消极关键词
        negative_keywords = ['生气', '愤怒', '糟糕', '失望', '垃圾', '差', '问题', '投诉']
        # 积极关键词
        positive_keywords = ['谢谢', '感谢', '很好', '满意', '不错', '喜欢', '棒']

        negative_count = sum(1 for keyword in negative_keywords if keyword in text)
        positive_count = sum(1 for keyword in positive_keywords if keyword in text)

        if negative_count > positive_count:
            return "消极"
        elif positive_count > negative_count:
            return "积极"
        else:
            return "中性"

    def is_faq_question(self, question: str) -> bool:
        """判断是否为FAQ问题"""
        for faq in self.faq_examples:
            if self._similarity(question, faq['question']) > 0.5:
                return True
        return False

    def _similarity(self, text1: str, text2: str) -> float:
        """简单的文本相似度计算"""
        words1 = set(text1)
        words2 = set(text2)
        intersection = words1 & words2
        union = words1 | words2
        return len(intersection) / len(union) if union else 0

    def handle_query(self, user_input: str) -> str:
        """处理用户查询"""
        # 检查是否询问姓名
        if not self.user_info['name'] and ('叫' in user_input or '名字' in user_input or '我是' in user_input):
            # 尝试提取姓名
            match = re.search(r'我是(.+)', user_input)
            if match:
                self.user_info['name'] = match.group(1).strip()

        # 分析情绪
        sentiment = self.analyze_sentiment(user_input)
        self.user_info['sentiment_history'].append(sentiment)

        # 记录问题
        self.user_info['questions_asked'].append({
            'question': user_input,
            'timestamp': datetime.now().isoformat(),
            'sentiment': sentiment
        })

        # 判断是否为FAQ
        if self.is_faq_question(user_input):
            response = self.faq_chain.run(input=user_input)
        else:
            response = self.chat_chain.run(
                input=user_input,
                sentiment=sentiment,
                user_name=self.user_info['name'] or "客户"
            )

        # 记录对话
        self.conversation_log.append({
            'timestamp': datetime.now().isoformat(),
            'user': user_input,
            'bot': response,
            'sentiment': sentiment
        })

        return response

    def create_ticket(self, issue_description: str) -> dict:
        """创建支持工单"""
        # 分析优先级
        sentiment = self.analyze_sentiment(issue_description)
        priority = "高" if sentiment == "消极" else "中"

        ticket = {
            'ticket_id': f"TK{datetime.now().strftime('%Y%m%d%H%M%S')}",
            'user_name': self.user_info['name'] or "未知",
            'issue': issue_description,
            'priority': priority,
            'status': '待处理',
            'created_at': datetime.now().isoformat(),
            'conversation_history': self.user_info['questions_asked']
        }

        # 保存工单
        ticket_file = f"tickets/ticket_{ticket['ticket_id']}.json"
        import os
        os.makedirs('tickets', exist_ok=True)

        with open(ticket_file, 'w', encoding='utf-8') as f:
            json.dump(ticket, f, ensure_ascii=False, indent=2)

        return ticket

    def save_conversation_log(self, filepath="conversation_log.json"):
        """保存对话日志"""
        log_data = {
            'user_info': self.user_info,
            'conversations': self.conversation_log,
            'saved_at': datetime.now().isoformat()
        }

        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(log_data, f, ensure_ascii=False, indent=2)

        print(f"✅ 对话日志已保存到 {filepath}")


def main():
    """主程序"""
    print("🤖 欢迎使用智能客服系统！\n")
    print("提示：请先告诉我您的姓名，例如：'我是张三'\n")

    bot = CustomerServiceBot()

    print("可用命令：")
    print("  - 直接输入问题进行咨询")
    print("  - /ticket <问题描述> - 创建支持工单")
    print("  - /save - 保存对话日志")
    print("  - /quit - 退出\n")

    while True:
        user_input = input("\n👤 您: ").strip()

        if not user_input:
            continue

        if user_input == "/quit":
            print("\n感谢使用！祝您生活愉快！👋")
            bot.save_conversation_log()
            break

        elif user_input.startswith("/ticket"):
            parts = user_input.split(maxsplit=1)
            if len(parts) > 1:
                ticket = bot.create_ticket(parts[1])
                print(f"\n✅ 工单已创建！")
                print(f"工单号：{ticket['ticket_id']}")
                print(f"优先级：{ticket['priority']}")
                print(f"状态：{ticket['status']}")
            else:
                print("\n请提供问题描述：/ticket <问题描述>")

        elif user_input == "/save":
            bot.save_conversation_log()

        else:
            response = bot.handle_query(user_input)
            print(f"\n🤖 客服: {response}")


if __name__ == "__main__":
    main()

"""Personalized Learning Assistant - Assignment 1"""
from langchain.prompts import ChatPromptTemplate
from langchain.memory import ConversationBufferMemory
from langchain_community.llms import Ollama
from langchain.chains import LLMChain
import json
from datetime import datetime

class LearningAssistant:
    """Personalized Learning Assistant"""

    def __init__(self, model_name="llama2"):
        """Initialize assistant"""
        self.llm = Ollama(model=model_name)
        self.memory = ConversationBufferMemory(
            memory_key="chat_history",
            return_messages=True
        )

        # Difficulty level configuration
        self.difficulty_levels = {
            "beginner": {
                "name": "Beginner",
                "description": "Use simple language, many examples, avoid technical jargon"
            },
            "intermediate": {
                "name": "Intermediate",
                "description": "Use standard terminology, provide moderate depth"
            },
            "advanced": {
                "name": "Advanced",
                "description": "Use professional terms, dive into technical details and principles"
            }
        }

        self.current_difficulty = "intermediate"
        self.learning_stats = {
            "topics_learned": [],
            "difficulty_changes": [],
            "total_interactions": 0
        }

        # Create prompt templates for different difficulty levels
        self.prompt_templates = self._create_prompt_templates()

    def _create_prompt_templates(self):
        """Create prompt templates for different difficulty levels"""
        templates = {}

        # Beginner template
        templates["beginner"] = ChatPromptTemplate.from_template("""
You are a patient tutor teaching a beginner student.

Student's question: {topic}

Please explain using the following approach:
1. Use simple, everyday language
2. Avoid technical jargon, or explain it in detail if necessary
3. Provide 2-3 concrete real-life examples
4. Use analogies to aid understanding
5. Keep it concise, under 200 words

Explanation:
""")

        # Intermediate template
        templates["intermediate"] = ChatPromptTemplate.from_template("""
You are a professional technical instructor teaching a student with some background.

Student's question: {topic}

Please explain using the following approach:
1. Use standard technical terminology
2. Provide clear concept definitions
3. Explain principles and how things work
4. Give 1-2 practical application scenarios
5. Be comprehensive but not overly deep

Explanation:
""")

        # Advanced template
        templates["advanced"] = ChatPromptTemplate.from_template("""
You are a senior technical expert communicating with an experienced developer.

Technical question: {topic}

Please explain using the following approach:
1. Use professional terminology and industry standards
2. Deep dive into technical principles and underlying implementation
3. Discuss performance, pros/cons, and best practices
4. Provide code examples or technical details
5. May include related advanced topics

Detailed explanation:
""")

        return templates

    def explain_topic(self, topic: str) -> str:
        """Explain topic based on current difficulty level"""
        # Get template for current difficulty
        template = self.prompt_templates[self.current_difficulty]

        # Create Chain
        chain = LLMChain(
            llm=self.llm,
            prompt=template,
            memory=self.memory
        )

        # Generate explanation
        response = chain.run(topic=topic)

        # Update statistics
        self.learning_stats["topics_learned"].append({
            "topic": topic,
            "difficulty": self.current_difficulty,
            "timestamp": datetime.now().isoformat()
        })
        self.learning_stats["total_interactions"] += 1

        return response

    def adjust_difficulty(self, feedback: str):
        """Adjust difficulty based on user feedback"""
        feedback = feedback.lower()

        old_difficulty = self.current_difficulty

        if "too easy" in feedback or "easy" in feedback or "简单" in feedback:
            if self.current_difficulty == "beginner":
                self.current_difficulty = "intermediate"
            elif self.current_difficulty == "intermediate":
                self.current_difficulty = "advanced"

        elif "too hard" in feedback or "hard" in feedback or "难" in feedback:
            if self.current_difficulty == "advanced":
                self.current_difficulty = "intermediate"
            elif self.current_difficulty == "intermediate":
                self.current_difficulty = "beginner"

        if old_difficulty != self.current_difficulty:
            self.learning_stats["difficulty_changes"].append({
                "from": old_difficulty,
                "to": self.current_difficulty,
                "timestamp": datetime.now().isoformat()
            })
            return f"Difficulty adjusted from {self.difficulty_levels[old_difficulty]['name']} to {self.difficulty_levels[self.current_difficulty]['name']}"

        return "Difficulty remains unchanged"

    def get_stats(self) -> dict:
        """Get learning statistics"""
        return {
            "current_difficulty": self.difficulty_levels[self.current_difficulty]['name'],
            "topics_count": len(self.learning_stats["topics_learned"]),
            "total_interactions": self.learning_stats["total_interactions"],
            "difficulty_changes": len(self.learning_stats["difficulty_changes"])
        }

    def save_progress(self, filepath="learning_progress.json"):
        """Save learning progress"""
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(self.learning_stats, f, ensure_ascii=False, indent=2)
        print(f"✅ Learning progress saved to {filepath}")

    def clear_memory(self):
        """Clear conversation memory"""
        self.memory.clear()
        print("✅ Conversation memory cleared")


def main():
    """Main program"""
    print("🎓 Welcome to the Personalized Learning Assistant!\n")

    assistant = LearningAssistant()

    print("Available commands:")
    print("  - Enter a question to learn")
    print("  - /difficulty [too easy/just right/too hard] - Adjust difficulty")
    print("  - /stats - View learning statistics")
    print("  - /save - Save learning progress")
    print("  - /clear - Clear conversation memory")
    print("  - /quit - Exit\n")

    while True:
        user_input = input("\n💭 Your question: ").strip()

        if not user_input:
            continue

        if user_input == "/quit":
            print("\n👋 Goodbye! Keep up the learning!")
            break

        elif user_input == "/stats":
            stats = assistant.get_stats()
            print("\n📊 Learning Statistics:")
            print(f"  Current difficulty: {stats['current_difficulty']}")
            print(f"  Topics learned: {stats['topics_count']}")
            print(f"  Total interactions: {stats['total_interactions']}")
            print(f"  Difficulty adjustments: {stats['difficulty_changes']}")

        elif user_input == "/save":
            assistant.save_progress()

        elif user_input == "/clear":
            assistant.clear_memory()

        elif user_input.startswith("/difficulty"):
            parts = user_input.split(maxsplit=1)
            if len(parts) > 1:
                result = assistant.adjust_difficulty(parts[1])
                print(f"\n{result}")
            else:
                print("\nPlease provide feedback: /difficulty [too easy/just right/too hard]")

        else:
            # Explain topic
            print(f"\n🎯 Current difficulty: {assistant.difficulty_levels[assistant.current_difficulty]['name']}")
            print("\n📖 Explanation:")
            response = assistant.explain_topic(user_input)
            print(response)
            print("\n💡 Tip: If the explanation is too easy or hard, use /difficulty to adjust")


if __name__ == "__main__":
    main()

"""Content Creation Pipeline System - Assignment 2"""
from langchain.prompts import ChatPromptTemplate
from langchain_community.llms import Ollama
from langchain.chains import LLMChain, SequentialChain
from datetime import datetime
import os

class ContentCreator:
    """Automated Content Creation System"""

    def __init__(self, model_name="llama2"):
        """Initialize creation system"""
        self.llm = Ollama(model=model_name)

        # Article type configuration
        self.article_types = {
            "technical": "Technical Article",
            "story": "Story/Fiction",
            "marketing": "Marketing Copy"
        }

        # Create output directory
        self.output_dir = "generated_content"
        os.makedirs(self.output_dir, exist_ok=True)

    def create_title_chain(self, article_type: str):
        """Create title generation Chain"""
        type_prompts = {
            "technical": "Generate an attractive technical article title. Requirements:\n1. Highlight technical points\n2. Include keywords\n3. Concise and powerful (10-15 words)",
            "story": "Generate a compelling story title. Requirements:\n1. Spark curiosity\n2. Be literary\n3. Concise and elegant (8-12 words)",
            "marketing": "Generate an eye-catching marketing title. Requirements:\n1. Highlight selling points\n2. Create urgency\n3. Trigger action (10-15 words)"
        }

        prompt = ChatPromptTemplate.from_template(f"""
{type_prompts[article_type]}

Topic: {{topic}}

Return only the title, nothing else.
Title:
""")

        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            output_key="title"
        )

    def create_outline_chain(self, article_type: str):
        """Create outline generation Chain"""
        type_prompts = {
            "technical": "Create detailed outline for technical article, including:\n1. Introduction\n2. Technical background\n3. Core content (3-5 points)\n4. Code example descriptions\n5. Summary",
            "story": "Create plot outline for story, including:\n1. Beginning\n2. Development\n3. Climax\n4. Resolution",
            "marketing": "Create structure for marketing copy, including:\n1. Pain point description\n2. Solution\n3. Product advantages\n4. Call to action"
        }

        prompt = ChatPromptTemplate.from_template(f"""
{type_prompts[article_type]}

Topic: {{topic}}
Title: {{title}}

Please generate a structured outline (use markdown format):
""")

        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            output_key="outline"
        )

    def create_content_chain(self, article_type: str):
        """Create content generation Chain"""
        type_prompts = {
            "technical": "Write technical article based on outline. Requirements:\n1. Professional and accurate language\n2. Include code examples\n3. Clear logic\n4. 800-1000 words",
            "story": "Create story based on outline. Requirements:\n1. Vivid plot\n2. Distinct characters\n3. Beautiful language\n4. 800-1000 words",
            "marketing": "Write marketing copy based on outline. Requirements:\n1. Persuasive language\n2. Highlight value\n3. Guide action\n4. 600-800 words"
        }

        prompt = ChatPromptTemplate.from_template(f"""
{type_prompts[article_type]}

Topic: {{topic}}
Title: {{title}}
Outline: {{outline}}

Please write the complete body content (use markdown format):
""")

        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            output_key="content"
        )

    def create_article(self, topic: str, article_type: str = "technical"):
        """Create complete article"""
        if article_type not in self.article_types:
            raise ValueError(f"Unsupported article type: {article_type}")

        print(f"\n📝 Creating {self.article_types[article_type]}...")
        print(f"Topic: {topic}\n")

        # Create three Chains
        title_chain = self.create_title_chain(article_type)
        outline_chain = self.create_outline_chain(article_type)
        content_chain = self.create_content_chain(article_type)

        # Combine into Sequential Chain
        overall_chain = SequentialChain(
            chains=[title_chain, outline_chain, content_chain],
            input_variables=["topic"],
            output_variables=["title", "outline", "content"],
            verbose=True
        )

        # Execute creation workflow
        print("🎯 Step 1: Generating title...")
        result = overall_chain({"topic": topic})

        # Display results
        print(f"\n✅ Title: {result['title']}\n")
        print("📋 Outline:")
        print(result['outline'])
        print("\n" + "="*50 + "\n")

        # Save to file
        filepath = self.save_article(
            topic=topic,
            title=result['title'],
            outline=result['outline'],
            content=result['content'],
            article_type=article_type
        )

        print(f"\n✅ Article saved to: {filepath}")

        return result

    def save_article(self, topic, title, outline, content, article_type):
        """Save article to file"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"{article_type}_{timestamp}.md"
        filepath = os.path.join(self.output_dir, filename)

        markdown_content = f"""# {title}

> Topic: {topic}
> Type: {self.article_types[article_type]}
> Generated: {datetime.now().strftime("%Y-%m-%d %H:%M:%S")}

---

## 📋 Article Outline

{outline}

---

## 📖 Content

{content}

---

*Auto-generated by AI*
"""

        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(markdown_content)

        return filepath


def main():
    """Main program"""
    print("✍️ Welcome to the AI Content Creation System!\n")

    creator = ContentCreator()

    print("Supported article types:")
    for key, value in creator.article_types.items():
        print(f"  {key}: {value}")
    print()

    while True:
        topic = input("Enter article topic (or 'quit' to exit): ").strip()

        if topic.lower() == 'quit':
            print("\n👋 Goodbye!")
            break

        if not topic:
            continue

        article_type = input("Select article type [technical/story/marketing]: ").strip() or "technical"

        try:
            creator.create_article(topic, article_type)
        except Exception as e:
            print(f"\n❌ Creation failed: {e}")

        print("\n" + "="*60 + "\n")


if __name__ == "__main__":
    main()

1. Smart Customer Service System (customer_service.py)

"""Smart Customer Service System - Assignment 3"""
from langchain.prompts import ChatPromptTemplate, FewShotChatMessagePromptTemplate
from langchain.memory import ConversationBufferMemory
from langchain_community.llms import Ollama
from langchain.chains import LLMChain
import json
from datetime import datetime
import re

class CustomerServiceBot:
    """Smart Customer Service Bot"""

    def __init__(self, model_name="llama2"):
        """Initialize customer service bot"""
        self.llm = Ollama(model=model_name)
        self.memory = ConversationBufferMemory(
            memory_key="chat_history",
            return_messages=True
        )

        # User information
        self.user_info = {
            "name": None,
            "questions_asked": [],
            "sentiment_history": []
        }

        # Conversation log
        self.conversation_log = []

        # FAQ examples (Few-shot)
        self.faq_examples = [
            {
                "question": "How do I reset my password?",
                "answer": "You can reset your password by following these steps:\n1. Click 'Forgot Password' on the login page\n2. Enter your registered email\n3. Check your email for reset link\n4. Click the link and set a new password"
            },
            {
                "question": "What is the return policy?",
                "answer": "Our return policy is as follows:\n- 30-day no-questions-asked returns\n- Items must be in original packaging\n- Non-quality issues require customer to pay shipping\n- 3-5 business days processing after return request"
            },
            {
                "question": "What payment methods are supported?",
                "answer": "We support multiple payment methods:\n- Credit/Debit cards\n- Alipay\n- WeChat Pay\n- PayPal\n- Bank transfer"
            },
            {
                "question": "How long does delivery take?",
                "answer": "Delivery timeframes:\n- Same city: 1-2 business days\n- Within province: 2-3 business days\n- Cross-province: 3-5 business days\n- Remote areas: 5-7 business days"
            },
            {
                "question": "How do I contact customer service?",
                "answer": "You can contact customer service via:\n- Hotline: 400-123-4567 (9:00-18:00)\n- Live chat: Click bottom-right icon\n- Email: support@company.com"
            }
        ]

        # Create Few-shot Prompt
        self.faq_chain = self._create_faq_chain()
        self.chat_chain = self._create_chat_chain()

    def _create_faq_chain(self):
        """Create FAQ Q&A Chain (Few-shot)"""
        # Few-shot example format
        example_prompt = ChatPromptTemplate.from_messages([
            ("human", "{question}"),
            ("ai", "{answer}")
        ])

        few_shot_prompt = FewShotChatMessagePromptTemplate(
            example_prompt=example_prompt,
            examples=self.faq_examples
        )

        # Complete Prompt
        final_prompt = ChatPromptTemplate.from_messages([
            ("system", "You are a professional customer service assistant. Please refer to the following FAQ examples to answer user questions."),
            few_shot_prompt,
            ("human", "{input}")
        ])

        return LLMChain(llm=self.llm, prompt=final_prompt)

    def _create_chat_chain(self):
        """Create conversation Chain"""
        template = """You are a friendly, professional customer service representative.

User sentiment: {sentiment}
User name: {user_name}

Please adjust your response style based on user sentiment:
- Negative sentiment: Show empathy, calm emotions, actively solve problems
- Neutral sentiment: Stay professional and friendly
- Positive sentiment: Stay enthusiastic, thank the user

Conversation history: {chat_history}

User question: {input}

Response:"""

        prompt = ChatPromptTemplate.from_template(template)
        return LLMChain(
            llm=self.llm,
            prompt=prompt,
            memory=self.memory
        )

    def analyze_sentiment(self, text: str) -> str:
        """Simple sentiment analysis (keyword-based)"""
        text = text.lower()

        # Negative keywords
        negative_keywords = ['angry', 'frustrated', 'terrible', 'disappointed', 'garbage', 'bad', 'problem', 'complaint']
        # Positive keywords
        positive_keywords = ['thank', 'thanks', 'good', 'satisfied', 'nice', 'like', 'great', 'excellent']

        negative_count = sum(1 for keyword in negative_keywords if keyword in text)
        positive_count = sum(1 for keyword in positive_keywords if keyword in text)

        if negative_count > positive_count:
            return "negative"
        elif positive_count > negative_count:
            return "positive"
        else:
            return "neutral"

    def is_faq_question(self, question: str) -> bool:
        """Check if it's a FAQ question"""
        for faq in self.faq_examples:
            if self._similarity(question, faq['question']) > 0.5:
                return True
        return False

    def _similarity(self, text1: str, text2: str) -> float:
        """Simple text similarity calculation"""
        words1 = set(text1.lower())
        words2 = set(text2.lower())
        intersection = words1 & words2
        union = words1 | words2
        return len(intersection) / len(union) if union else 0

    def handle_query(self, user_input: str) -> str:
        """Handle user query"""
        # Check if asking for name
        if not self.user_info['name'] and ('name is' in user_input.lower() or 'i am' in user_input.lower() or "i'm" in user_input.lower()):
            # Try to extract name
            match = re.search(r"(?:i am|i'm|name is)\s+(\w+)", user_input, re.IGNORECASE)
            if match:
                self.user_info['name'] = match.group(1).strip()

        # Analyze sentiment
        sentiment = self.analyze_sentiment(user_input)
        self.user_info['sentiment_history'].append(sentiment)

        # Record question
        self.user_info['questions_asked'].append({
            'question': user_input,
            'timestamp': datetime.now().isoformat(),
            'sentiment': sentiment
        })

        # Check if FAQ
        if self.is_faq_question(user_input):
            response = self.faq_chain.run(input=user_input)
        else:
            response = self.chat_chain.run(
                input=user_input,
                sentiment=sentiment,
                user_name=self.user_info['name'] or "Customer"
            )

        # Log conversation
        self.conversation_log.append({
            'timestamp': datetime.now().isoformat(),
            'user': user_input,
            'bot': response,
            'sentiment': sentiment
        })

        return response

    def create_ticket(self, issue_description: str) -> dict:
        """Create support ticket"""
        # Analyze priority
        sentiment = self.analyze_sentiment(issue_description)
        priority = "High" if sentiment == "negative" else "Medium"

        ticket = {
            'ticket_id': f"TK{datetime.now().strftime('%Y%m%d%H%M%S')}",
            'user_name': self.user_info['name'] or "Unknown",
            'issue': issue_description,
            'priority': priority,
            'status': 'Pending',
            'created_at': datetime.now().isoformat(),
            'conversation_history': self.user_info['questions_asked']
        }

        # Save ticket
        ticket_file = f"tickets/ticket_{ticket['ticket_id']}.json"
        import os
        os.makedirs('tickets', exist_ok=True)

        with open(ticket_file, 'w', encoding='utf-8') as f:
            json.dump(ticket, f, ensure_ascii=False, indent=2)

        return ticket

    def save_conversation_log(self, filepath="conversation_log.json"):
        """Save conversation log"""
        log_data = {
            'user_info': self.user_info,
            'conversations': self.conversation_log,
            'saved_at': datetime.now().isoformat()
        }

        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(log_data, f, ensure_ascii=False, indent=2)

        print(f"✅ Conversation log saved to {filepath}")


def main():
    """Main program"""
    print("🤖 Welcome to Smart Customer Service System!\n")
    print("Tip: Please tell me your name first, e.g., 'I am John'\n")

    bot = CustomerServiceBot()

    print("Available commands:")
    print("  - Enter a question for assistance")
    print("  - /ticket  - Create support ticket")
    print("  - /save - Save conversation log")
    print("  - /quit - Exit\n")

    while True:
        user_input = input("\n👤 You: ").strip()

        if not user_input:
            continue

        if user_input == "/quit":
            print("\nThank you for using our service! Have a great day! 👋")
            bot.save_conversation_log()
            break

        elif user_input.startswith("/ticket"):
            parts = user_input.split(maxsplit=1)
            if len(parts) > 1:
                ticket = bot.create_ticket(parts[1])
                print(f"\n✅ Ticket created!")
                print(f"Ticket ID: {ticket['ticket_id']}")
                print(f"Priority: {ticket['priority']}")
                print(f"Status: {ticket['status']}")
            else:
                print("\nPlease provide issue description: /ticket ")

        elif user_input == "/save":
            bot.save_conversation_log()

        else:
            response = bot.handle_query(user_input)
            print(f"\n🤖 Agent: {response}")


if __name__ == "__main__":
    main()

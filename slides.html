import tiktoken
import matplotlib.pyplot as plt

class TokenEconomicsDemo:
    def __init__(self):
        self.encoder = tiktoken.get_encoding("cl100k_base")

    def compare_languages(self):
        """对比不同语言的Token效率"""

        test_cases = {
            'English': "Artificial Intelligence is transforming the world",
            '中文': "人工智能正在改变世界",
            '日本語': "人工知能は世界を変えています",
            'Code': "def hello(): return 'world'",
            'Mixed': "AI(人工智能) is 改变ing the world"
        }

        print("🌍 多语言Token效率对比")
        print("-" * 50)

        results = []
        for lang, text in test_cases.items():
            tokens = self.encoder.encode(text)
            char_count = len(text)
            token_count = len(tokens)
            efficiency = char_count / token_count

            print(f"\n{lang}:")
            print(f"  文本: {text}")
            print(f"  字符数: {char_count}")
            print(f"  Token数: {token_count}")
            print(f"  效率比: {efficiency:.2f}")

            results.append({
                'Language': lang,
                'Tokens': token_count,
                'Efficiency': efficiency
            })

        return results

# 成本计算实战
def calculate_cost(text, model='gpt-3.5-turbo', calls_per_day=1000):
    """实时计算API成本"""

    encoder = tiktoken.get_encoding("cl100k_base")
    tokens = len(encoder.encode(text))

    # 价格表（2025年最新）
    pricing = {
        'gpt-5': 0.05,  # $0.05/1K tokens输入
        'gpt-4.5-turbo': 0.015,
        'gpt-4o': 0.0075,
        'gpt-4o-mini': 0.0004,
        'gpt-3.5-turbo': 0.0005,
        'claude-sonnet-4.5': 0.003,
        'gemini-2.0-pro': 0.002
    }

    cost_per_call = (tokens / 1000) * pricing[model]
    daily_cost = cost_per_call * calls_per_day
    monthly_cost = daily_cost * 30

    print(f"📝 Token数: {tokens}")
    print(f"💵 单次成本: ${cost_per_call:.6f}")
    print(f"📅 日成本({calls_per_day}次): ${daily_cost:.2f}")
    print(f"📆 月成本: ${monthly_cost:.2f}")
    print(f"🎯 年成本: ${monthly_cost*12:.2f}")

    return monthly_cost

🎛️ Temperature参数实验

Temperature参数解释

Temperature控制输出的随机性：

  • Temperature = 0：严肃的会计师，永远准确
  • Temperature = 1：创意总监，天马行空
  • Temperature = 2：喝醉的诗人，完全随机
Temperature: 0.0-0.3

✅ 适用于：

  • 数学计算
  • 事实查询
  • 代码生成
Temperature: 0.5-0.8

✅ 适用于：

  • 客服对话
  • 文章写作
  • 通用对话
Temperature: 0.9-2.0

✅ 适用于：

  • 创意写作
  • 头脑风暴
  • 艺术创作

# parameter_lab.py
import openai
import time
from typing import List

class ParameterLab:
    """参数实验室"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.results = []

    def experiment_temperature(self,
                              prompt: str,
                              temperatures: List[float] = [0, 0.5, 1.0, 1.5],
                              n_samples: int = 3):
        """Temperature实验"""

        print(f"🧪 Temperature实验")
        print(f"Prompt: {prompt}")
        print("=" * 60)

        for temp in temperatures:
            print(f"\n🌡️ Temperature = {temp}")
            print("-" * 40)

            samples = []
            for i in range(n_samples):
                response = self.client.chat.completions.create(
                    model="gpt-3.5-turbo",
                    messages=[{"role": "user", "content": prompt}],
                    temperature=temp,
                    max_tokens=30
                )

                text = response.choices[0].message.content
                samples.append(text)
                print(f"  样本{i+1}: {text}")

            # 计算多样性
            diversity = self._calculate_diversity(samples)
            print(f"  📊 多样性得分: {diversity:.2f}")

    def _calculate_diversity(self, texts: List[str]) -> float:
        """计算文本多样性"""
        if len(texts) < 2:
            return 0.0

        all_words = []
        for text in texts:
            all_words.extend(text.split())

        unique_words = set(all_words)
        diversity = len(unique_words) / len(all_words) if all_words else 0

        return diversity

# 使用示例

    # 测试创意写作

    # 测试事实查询
    lab.experiment_temperature("中国的首都是", temperatures=[0, 0.5, 1.0])

# model_benchmark.py
import openai
import time
import pandas as pd
from datetime import datetime

class ModelBenchmark:
    """模型基准测试系统"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.test_suites = {
            'creative': {
                'name': '创意写作',
                'prompts': [
                    "写一个六字微小说",
                    "为咖啡店起3个创意名字",
                    "用emoji讲一个故事"
                ],
                'best_temp': 0.9
            },
            'analytical': {
                'name': '分析推理',
                'prompts': [
                    "分析这个数列的规律：2, 4, 8, 16, ?",
                    "如果所有玫瑰都是花，那么玫瑰需要什么？",
                ],
                'best_temp': 0.3
            },
            'coding': {
                'name': '代码生成',
                'prompts': [
                    "写一个Python函数检查素数",
                    "用一行代码反转字符串",
                ],
                'best_temp': 0.2
            }
        }
        self.results = []

    def benchmark_models(self, models=['gpt-3.5-turbo', 'gpt-4o-mini']):
        """对比测试多个模型"""

        print("🏁 开始模型基准测试")
        print("=" * 70)

        for suite_name, suite in self.test_suites.items():
            print(f"\n📋 测试套件: {suite['name']}")
            print("-" * 70)

            for model in models:
                print(f"\n  🤖 测试模型: {model}")

                total_time = 0
                total_tokens = 0
                total_cost = 0

                for prompt in suite['prompts']:
                    start_time = time.time()

                    try:
                        response = self.client.chat.completions.create(
                            model=model,
                            messages=[{"role": "user", "content": prompt}],
                            temperature=suite['best_temp'],
                            max_tokens=100
                        )

                        elapsed = time.time() - start_time
                        tokens = response.usage.total_tokens
                        cost = self._calculate_cost(model, tokens)

                        total_time += elapsed
                        total_tokens += tokens
                        total_cost += cost

                        print(f"    ✓ {prompt[:30]}...")
                        print(f"      时间: {elapsed:.2f}s | Tokens: {tokens}")

                    except Exception as e:
                        print(f"    ✗ 错误: {str(e)}")

                # 汇总统计
                avg_time = total_time / len(suite['prompts'])
                avg_tokens = total_tokens / len(suite['prompts'])

                print(f"\n    📊 统计:")
                print(f"      平均响应: {avg_time:.2f}s")
                print(f"      平均Token: {avg_tokens:.0f}")
                print(f"      总成本: ${total_cost:.4f}")

                self.results.append({
                    'suite': suite['name'],
                    'model': model,
                    'avg_time': avg_time,
                    'avg_tokens': avg_tokens,
                    'total_cost': total_cost
                })

    def _calculate_cost(self, model, tokens):
        """计算成本"""
        pricing = {
            'gpt-3.5-turbo': 0.001,
            'gpt-4o-mini': 0.0006,
            'gpt-4o': 0.01
        }
        price_per_1k = pricing.get(model, 0.001)
        return (tokens / 1000) * price_per_1k

    def generate_report(self):
        """生成对比报告"""
        df = pd.DataFrame(self.results)

        print("\n" + "=" * 70)
        print("📊 模型对比报告")
        print("=" * 70)

        for suite in df['suite'].unique():
            suite_df = df[df['suite'] == suite]
            print(f"\n### {suite}")
            print("-" * 40)

            for _, row in suite_df.iterrows():
                print(f"  {row['model']:20} | "
                      f"⏱️ {row['avg_time']:.2f}s | "
                      f"🔤 {row['avg_tokens']:.0f} tokens | "
                      f"💰 ${row['total_cost']:.4f}")

        return df

# 运行基准测试
if __name__ == "__main__":
    benchmark = ModelBenchmark()
    benchmark.benchmark_models(['gpt-3.5-turbo', 'gpt-4o-mini'])
    report = benchmark.generate_report()

    # 保存结果
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    report.to_csv(f'benchmark_{timestamp}.csv', index=False)

# smart_customer_service.py
import openai
import sqlite3
from typing import Dict

class SmartCustomerService:
    """智能客服系统 - 自动选择模型和参数"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.faq = {
            '退货': {
                'keywords': ['退货', '退款', '退换', '不满意'],
                'response': '我们提供30天无理由退换货服务。',
                'needs_ai': False
            },
            '物流': {
                'keywords': ['快递', '物流', '发货', '配送'],
                'response': '一般订单24小时内发货，3-5个工作日送达。',
                'needs_ai': False
            },
            '技术支持': {
                'keywords': ['不会用', '怎么用', '故障'],
                'needs_ai': True
            }
        }

    def detect_intent(self, message: str) -> str:
        """意图识别"""
        for intent, config in self.faq.items():
            for keyword in config['keywords']:
                if keyword in message.lower():
                    return intent
        return 'general'

    def select_model_and_params(self, intent: str, message_length: int) -> Dict:
        """根据意图选择最佳模型和参数"""

        if intent == 'general':
            # 一般咨询，使用便宜快速的模型
            return {
                'model': 'gpt-3.5-turbo',
                'temperature': 0.7,
                'max_tokens': 150,
                'system_prompt': '你是一个友好专业的客服代表。'
            }
        elif intent == '技术支持':
            # 技术问题，需要准确性
            return {
                'model': 'gpt-4o-mini',
                'temperature': 0.3,
                'max_tokens': 300,
                'system_prompt': '你是一个技术支持专家。'
            }
        elif message_length > 200:
            # 长消息，可能是复杂问题
            return {
                'model': 'gpt-4o-mini',
                'temperature': 0.5,
                'max_tokens': 400,
                'system_prompt': '你是一个高级客服专员。'
            }
        else:
            return {
                'model': 'gpt-3.5-turbo',
                'temperature': 0.6,
                'max_tokens': 100,
                'system_prompt': '你是一个客服助手。'
            }

    def handle_message(self, user_id: str, message: str) -> Dict:
        """处理用户消息"""

        print(f"\n📨 收到消息: {message}")
        print("-" * 50)

        # 1. 意图识别
        intent = self.detect_intent(message)
        print(f"🎯 识别意图: {intent}")

        # 2. 检查FAQ
        if intent in self.faq and not self.faq[intent]['needs_ai']:
            response = self.faq[intent]['response']
            print(f"📋 使用预设回答")

            return {
                'response': response,
                'model_used': 'FAQ',
                'tokens_used': 0,
                'cost': 0
            }

        # 3. 选择模型和参数
        config = self.select_model_and_params(intent, len(message))
        print(f"🤖 选择模型: {config['model']}")

        # 4. 调用AI
        try:
            response_obj = self.client.chat.completions.create(
                model=config['model'],
                messages=[
                    {"role": "system", "content": config['system_prompt']},
                    {"role": "user", "content": message}
                ],
                temperature=config['temperature'],
                max_tokens=config['max_tokens']
            )

            response = response_obj.choices[0].message.content
            tokens = response_obj.usage.total_tokens
            cost = (tokens / 1000) * 0.001

            print(f"\n💬 回复: {response[:50]}...")
            print(f"💰 成本: ${cost:.6f}")

            return {
                'response': response,
                'model_used': config['model'],
                'tokens_used': tokens,
                'cost': cost
            }

        except Exception as e:
            return {
                'response': "抱歉，系统暂时无法处理。",
                'model_used': 'error',
                'tokens_used': 0,
                'cost': 0
            }

# content_creation_assistant.py
from enum import Enum
from typing import List, Dict

class ContentType(Enum):
    BLOG_POST = "blog_post"
    SOCIAL_MEDIA = "social_media"
    PRODUCT_DESC = "product_description"

class ContentCreationAssistant:
    """内容创作助手"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.templates = {
            ContentType.BLOG_POST: {
                'structure': ['标题', '导言', '主体', '结论'],
                'tone': 'professional',
                'length': 800,
                'model': 'gpt-4o-mini',
                'temperature': 0.7
            },
            ContentType.SOCIAL_MEDIA: {
                'structure': ['Hook', '内容', 'Hashtags'],
                'tone': 'casual',
                'length': 280,
                'model': 'gpt-3.5-turbo',
                'temperature': 0.9
            },
            ContentType.PRODUCT_DESC: {
                'structure': ['特点', '优势', '场景'],
                'tone': 'persuasive',
                'length': 200,
                'model': 'gpt-3.5-turbo',
                'temperature': 0.6
            }
        }

    def create_content(self,
                      content_type: ContentType,
                      topic: str,
                      keywords: List[str] = None) -> Dict:
        """创建内容"""

        template = self.templates[content_type]

        # 构建prompt
        prompt = self._build_prompt(
            content_type,
            topic,
            keywords,
            template['tone'],
            template['structure'],
            template['length']
        )

        print(f"\n✍️ 创建 {content_type.value}")
        print(f"📝 主题: {topic}")
        print(f"🤖 模型: {template['model']}")
        print(f"🌡️ Temperature: {template['temperature']}")

        # 生成内容
        response = self.client.chat.completions.create(
            model=template['model'],
            messages=[
                {"role": "system", "content": "你是专业内容创作者"},
                {"role": "user", "content": prompt}
            ],
            temperature=template['temperature'],
            max_tokens=template['length'] * 2
        )

        content = response.choices[0].message.content
        tokens = response.usage.total_tokens
        cost = (tokens / 1000) * 0.001

        # 质量评分
        quality = self._evaluate_quality(content, keywords)

        print(f"\n📊 生成完成:")
        print(f"  长度: {len(content)}字符")
        print(f"  质量: {quality:.1f}/10")
        print(f"  成本: ${cost:.4f}")

        return {
            'content': content,
            'quality_score': quality,
            'cost': cost
        }

    def _build_prompt(self, content_type, topic, keywords,
                     tone, structure, length) -> str:
        """构建优化的prompt"""

        parts = [
            f"创建一篇关于'{topic}'的{content_type.value}。",
            f"\n要求:",
            f"1. 语气: {tone}",
            f"2. 长度: 约{length}字",
            f"3. 结构: {', '.join(structure)}"
        ]

        if keywords:
            parts.append(f"4. 关键词: {', '.join(keywords)}")

        parts.append("\n请直接输出内容。")

        return '\n'.join(parts)

    def _evaluate_quality(self, content: str,
                         keywords: List[str]) -> float:
        """评估内容质量"""
        score = 10.0

        if len(content) < 50:
            score -= 2

        if keywords:
            covered = sum(1 for kw in keywords
                         if kw.lower() in content.lower())
            score *= (covered / len(keywords))

        return min(max(score, 0), 10)

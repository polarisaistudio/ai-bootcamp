什么是Token？

想象Token就像乐高积木：

  • 英文是标准积木，一个词一块
  • 中文是特殊积木，一个字可能要2-3块
  • 这直接影响你的成本！

class Config:
    """集中管理配置"""
    
    # API Keys
    OPENAI_API_KEY = os.getenv('OPENAI_API_KEY')
    ANTHROPIC_API_KEY = os.getenv('ANTHROPIC_API_KEY')
    
    @classmethod
    def validate(cls):
        if not cls.OPENAI_API_KEY:
            raise ValueError("❌ 请设置OPENAI_API_KEY")
        print("✅ OpenAI密钥已配置")

# 测试配置
if __name__ == "__main__":
    Config.validate()

🔌 第一个API调用

🎉 让我们写第一个程序！

# first_api_call.py
import openai
from config import Config
import json

def my_first_chat():
    """我的第一个AI对话"""
    
    # 1. 设置客户端
    client = openai.OpenAI(api_key=Config.OPENAI_API_KEY)
    
    # 2. 准备消息
    messages = [
        {"role": "system", "content": "你是一个友好的AI助手"},
        {"role": "user", "content": "你好！请用一句话介绍自己。"}
    ]
    
    print("🤖 正在调用OpenAI API...")
    print("-" * 50)
    
    # 3. 调用API
    try:
        response = client.chat.completions.create(
            model="gpt-3.5-turbo",
            messages=messages,
            temperature=0.7,
            max_tokens=50
        )
        
        # 4. 提取回复
        ai_message = response.choices[0].message.content
        
        # 5. 显示结果
        print(f"AI回复: {ai_message}")
        print("-" * 50)
        
        # 6. 显示使用统计
        print(f"📊 使用统计:")
        print(f"  - 模型: {response.model}")
        print(f"  - 输入Token: {response.usage.prompt_tokens}")
        print(f"  - 输出Token: {response.usage.completion_tokens}")
        print(f"  - 总Token: {response.usage.total_tokens}")
        
        # 7. 计算成本
        cost = (response.usage.total_tokens / 1000) * 0.001
        print(f"  - 成本: ${cost:.6f}")
        
    except Exception as e:
        print(f"❌ 错误: {e}")

if __name__ == "__main__":
    my_first_chat()

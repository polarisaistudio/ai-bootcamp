# token_economics.py
import tiktoken
import matplotlib.pyplot as plt

class TokenEconomicsDemo:
    def __init__(self):
        self.encoder = tiktoken.get_encoding("cl100k_base")

    def compare_languages(self):
        """Compare Token efficiency across different languages"""

        test_cases = {
            'English': "Artificial Intelligence is transforming the world",
            'Chinese': "人工智能正在改变世界",
            'Japanese': "人工知能は世界を変えています",
            'Code': "def hello(): return 'world'",
            'Mixed': "AI(人工智能) is 改变ing the world"
        }

        print("🌍 Multi-language Token Efficiency Comparison")
        print("-" * 50)

        results = []
        for lang, text in test_cases.items():
            tokens = self.encoder.encode(text)
            char_count = len(text)
            token_count = len(tokens)
            efficiency = char_count / token_count

            print(f"\n{lang}:")
            print(f"  Text: {text}")
            print(f"  Characters: {char_count}")
            print(f"  Tokens: {token_count}")
            print(f"  Efficiency ratio: {efficiency:.2f}")

            results.append({
                'Language': lang,
                'Tokens': token_count,
                'Efficiency': efficiency
            })

        return results

# Cost calculation practice
def calculate_cost(text, model='gpt-3.5-turbo', calls_per_day=1000):
    """Real-time API cost calculation"""

    encoder = tiktoken.get_encoding("cl100k_base")
    tokens = len(encoder.encode(text))

    # Pricing table (2025 latest)
    pricing = {
        'gpt-5': 0.05,  # $0.05/1K tokens input
        'gpt-4.5-turbo': 0.015,
        'gpt-4o': 0.0075,
        'gpt-4o-mini': 0.0004,
        'gpt-3.5-turbo': 0.0005,
        'claude-sonnet-4.5': 0.003,
        'gemini-2.0-pro': 0.002
    }

    cost_per_call = (tokens / 1000) * pricing[model]
    daily_cost = cost_per_call * calls_per_day
    monthly_cost = daily_cost * 30

    print(f"📝 Token count: {tokens}")
    print(f"💵 Cost per call: ${cost_per_call:.6f}")
    print(f"📅 Daily cost ({calls_per_day} calls): ${daily_cost:.2f}")
    print(f"📆 Monthly cost: ${monthly_cost:.2f}")
    print(f"🎯 Yearly cost: ${monthly_cost*12:.2f}")

    return monthly_cost

# parameter_lab.py
import openai
import time
from typing import List

class ParameterLab:
    """Parameter Laboratory"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.results = []

    def experiment_temperature(self,
                              prompt: str,
                              temperatures: List[float] = [0, 0.5, 1.0, 1.5],
                              n_samples: int = 3):
        """Temperature experiment"""

        print(f"🧪 Temperature Experiment")
        print(f"Prompt: {prompt}")
        print("=" * 60)

        for temp in temperatures:
            print(f"\n🌡️ Temperature = {temp}")
            print("-" * 40)

            samples = []
            for i in range(n_samples):
                response = self.client.chat.completions.create(
                    model="gpt-3.5-turbo",
                    messages=[{"role": "user", "content": prompt}],
                    temperature=temp,
                    max_tokens=30
                )

                text = response.choices[0].message.content
                samples.append(text)
                print(f"  Sample {i+1}: {text}")

            # Calculate diversity
            diversity = self._calculate_diversity(samples)
            print(f"  📊 Diversity score: {diversity:.2f}")

    def _calculate_diversity(self, texts: List[str]) -> float:
        """Calculate text diversity"""
        if len(texts) < 2:
            return 0.0

        all_words = []
        for text in texts:
            all_words.extend(text.split())

        unique_words = set(all_words)
        diversity = len(unique_words) / len(all_words) if all_words else 0

        return diversity

# model_benchmark.py
import openai
import time
import pandas as pd
from datetime import datetime

class ModelBenchmark:
    """Model Benchmarking System"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.test_suites = {
            'creative': {
                'name': 'Creative Writing',
                'prompts': [
                    "Write a six-word micro story",
                    "Create 3 creative names for a coffee shop",
                    "Tell a story using emojis"
                ],
                'best_temp': 0.9
            },
            'analytical': {
                'name': 'Analysis & Reasoning',
                'prompts': [
                    "Analyze the pattern in this sequence: 2, 4, 8, 16, ?",
                    "If all roses are flowers, what do roses need?",
                ],
                'best_temp': 0.3
            },
            'coding': {
                'name': 'Code Generation',
                'prompts': [
                    "Write a Python function to check for prime numbers",
                    "Reverse a string in one line of code",
                ],
                'best_temp': 0.2
            }
        }
        self.results = []

    def benchmark_models(self, models=['gpt-3.5-turbo', 'gpt-4o-mini']):
        """Benchmark multiple models"""

        print("🏁 Starting Model Benchmark Test")
        print("=" * 70)

        for suite_name, suite in self.test_suites.items():
            print(f"\n📋 Test Suite: {suite['name']}")
            print("-" * 70)

            for model in models:
                print(f"\n  🤖 Testing Model: {model}")

                total_time = 0
                total_tokens = 0
                total_cost = 0

                for prompt in suite['prompts']:
                    start_time = time.time()

                    try:
                        response = self.client.chat.completions.create(
                            model=model,
                            messages=[{"role": "user", "content": prompt}],
                            temperature=suite['best_temp'],
                            max_tokens=100
                        )

                        elapsed = time.time() - start_time
                        tokens = response.usage.total_tokens
                        cost = self._calculate_cost(model, tokens)

                        total_time += elapsed
                        total_tokens += tokens
                        total_cost += cost

                        print(f"    ✓ {prompt[:30]}...")
                        print(f"      Time: {elapsed:.2f}s | Tokens: {tokens}")

                    except Exception as e:
                        print(f"    ✗ Error: {str(e)}")

                # Summary statistics
                avg_time = total_time / len(suite['prompts'])
                avg_tokens = total_tokens / len(suite['prompts'])

                print(f"\n    📊 Statistics:")
                print(f"      Avg response: {avg_time:.2f}s")
                print(f"      Avg tokens: {avg_tokens:.0f}")
                print(f"      Total cost: ${total_cost:.4f}")

                self.results.append({
                    'suite': suite['name'],
                    'model': model,
                    'avg_time': avg_time,
                    'avg_tokens': avg_tokens,
                    'total_cost': total_cost
                })

    def _calculate_cost(self, model, tokens):
        """Calculate cost"""
        pricing = {
            'gpt-3.5-turbo': 0.001,
            'gpt-4o-mini': 0.0006,
            'gpt-4o': 0.01
        }
        price_per_1k = pricing.get(model, 0.001)
        return (tokens / 1000) * price_per_1k

    def generate_report(self):
        """Generate comparison report"""
        df = pd.DataFrame(self.results)

        print("\n" + "=" * 70)
        print("📊 Model Comparison Report")
        print("=" * 70)

        for suite in df['suite'].unique():
            suite_df = df[df['suite'] == suite]
            print(f"\n### {suite}")
            print("-" * 40)

            for _, row in suite_df.iterrows():
                print(f"  {row['model']:20} | "
                      f"⏱️ {row['avg_time']:.2f}s | "
                      f"🔤 {row['avg_tokens']:.0f} tokens | "
                      f"💰 ${row['total_cost']:.4f}")

        return df

# Run benchmark
if __name__ == "__main__":
    benchmark = ModelBenchmark()
    benchmark.benchmark_models(['gpt-3.5-turbo', 'gpt-4o-mini'])
    report = benchmark.generate_report()

    # Save results
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    report.to_csv(f'benchmark_{timestamp}.csv', index=False)

# smart_customer_service.py
import openai
import sqlite3
from typing import Dict

class SmartCustomerService:
    """Smart Customer Service System - Automatic Model and Parameter Selection"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.faq = {
            'returns': {
                'keywords': ['return', 'refund', 'exchange', 'unsatisfied'],
                'response': 'We offer 30-day no-questions-asked return service.',
                'needs_ai': False
            },
            'shipping': {
                'keywords': ['shipping', 'delivery', 'logistics', 'when'],
                'response': 'Orders usually ship within 24 hours, arriving in 3-5 business days.',
                'needs_ai': False
            },
            'technical': {
                'keywords': ['how to use', 'tutorial', 'problem', 'error'],
                'needs_ai': True
            }
        }

    def detect_intent(self, message: str) -> str:
        """Intent detection"""
        for intent, config in self.faq.items():
            for keyword in config['keywords']:
                if keyword in message.lower():
                    return intent
        return 'general'

    def select_model_and_params(self, intent: str, message_length: int) -> Dict:
        """Select best model and parameters based on intent"""

        if intent == 'general':
            # General inquiry, use cheap fast model
            return {
                'model': 'gpt-3.5-turbo',
                'temperature': 0.7,
                'max_tokens': 150,
                'system_prompt': 'You are a friendly professional customer service representative.'
            }
        elif intent == 'technical':
            # Technical questions need accuracy
            return {
                'model': 'gpt-4o-mini',
                'temperature': 0.3,
                'max_tokens': 300,
                'system_prompt': 'You are a technical support expert.'
            }
        elif message_length > 200:
            # Long message, possibly complex issue
            return {
                'model': 'gpt-4o-mini',
                'temperature': 0.5,
                'max_tokens': 400,
                'system_prompt': 'You are a senior customer service specialist.'
            }
        else:
            return {
                'model': 'gpt-3.5-turbo',
                'temperature': 0.6,
                'max_tokens': 100,
                'system_prompt': 'You are a customer service assistant.'
            }

    def handle_message(self, user_id: str, message: str) -> Dict:
        """Handle user message"""

        print(f"\n📨 Received message: {message}")
        print("-" * 50)

        # 1. Intent detection
        intent = self.detect_intent(message)
        print(f"🎯 Detected intent: {intent}")

        # 2. Check FAQ
        if intent in self.faq and not self.faq[intent]['needs_ai']:
            response = self.faq[intent]['response']
            print(f"📋 Using preset answer")

            return {
                'response': response,
                'model_used': 'FAQ',
                'tokens_used': 0,
                'cost': 0
            }

        # 3. Select model and parameters
        config = self.select_model_and_params(intent, len(message))
        print(f"🤖 Selected model: {config['model']}")

        # 4. Call AI
        try:
            response_obj = self.client.chat.completions.create(
                model=config['model'],
                messages=[
                    {"role": "system", "content": config['system_prompt']},
                    {"role": "user", "content": message}
                ],
                temperature=config['temperature'],
                max_tokens=config['max_tokens']
            )

            response = response_obj.choices[0].message.content
            tokens = response_obj.usage.total_tokens
            cost = (tokens / 1000) * 0.001

            print(f"\n💬 Reply: {response[:50]}...")
            print(f"💰 Cost: ${cost:.6f}")

            return {
                'response': response,
                'model_used': config['model'],
                'tokens_used': tokens,
                'cost': cost
            }

        except Exception as e:
            return {
                'response': "Sorry, the system is temporarily unavailable.",
                'model_used': 'error',
                'tokens_used': 0,
                'cost': 0
            }

# content_creation_assistant.py
from enum import Enum
from typing import List, Dict

class ContentType(Enum):
    """Content type enumeration"""
    BLOG_POST = "blog_post"
    SOCIAL_MEDIA = "social_media"
    PRODUCT_DESC = "product_description"

class ContentCreationAssistant:
    """Content Creation Assistant"""

    def __init__(self):
        self.client = openai.OpenAI()
        self.templates = {
            ContentType.BLOG_POST: {
                'structure': ['Title', 'Introduction', 'Body', 'Conclusion'],
                'tone': 'professional',
                'length': 800,
                'model': 'gpt-4o-mini',
                'temperature': 0.7
            },
            ContentType.SOCIAL_MEDIA: {
                'structure': ['Hook', 'Content', 'Hashtags'],
                'tone': 'casual',
                'length': 280,
                'model': 'gpt-3.5-turbo',
                'temperature': 0.9
            },
            ContentType.PRODUCT_DESC: {
                'structure': ['Features', 'Benefits', 'Scenarios'],
                'tone': 'persuasive',
                'length': 200,
                'model': 'gpt-3.5-turbo',
                'temperature': 0.6
            }
        }

    def create_content(self,
                      content_type: ContentType,
                      topic: str,
                      keywords: List[str] = None) -> Dict:
        """Create content"""

        template = self.templates[content_type]

        # Build prompt
        prompt = self._build_prompt(
            content_type,
            topic,
            keywords,
            template['tone'],
            template['structure'],
            template['length']
        )

        print(f"\n✍️ Creating {content_type.value}")
        print(f"📝 Topic: {topic}")
        print(f"🤖 Model: {template['model']}")
        print(f"🌡️ Temperature: {template['temperature']}")

        # Generate content
        response = self.client.chat.completions.create(
            model=template['model'],
            messages=[
                {"role": "system", "content": "You are a professional content creator"},
                {"role": "user", "content": prompt}
            ],
            temperature=template['temperature'],
            max_tokens=template['length'] * 2
        )

        content = response.choices[0].message.content
        tokens = response.usage.total_tokens
        cost = (tokens / 1000) * 0.001

        # Quality score
        quality = self._evaluate_quality(content, keywords)

        print(f"\n📊 Generation complete:")
        print(f"  Length: {len(content)} characters")
        print(f"  Quality: {quality:.1f}/10")
        print(f"  Cost: ${cost:.4f}")

        return {
            'content': content,
            'quality_score': quality,
            'cost': cost
        }

    def _build_prompt(self, content_type, topic, keywords,
                     tone, structure, length) -> str:
        """Build optimized prompt"""

        parts = [
            f"Create a {content_type.value} about '{topic}'.",
            f"\nRequirements:",
            f"1. Tone: {tone}",
            f"2. Length: approximately {length} words",
            f"3. Structure: {', '.join(structure)}"
        ]

        if keywords:
            parts.append(f"4. Keywords: {', '.join(keywords)}")

        parts.append("\nPlease output the content directly.")

        return '\n'.join(parts)

    def _evaluate_quality(self, content: str,
                         keywords: List[str]) -> float:
        """Evaluate content quality"""
        score = 10.0

        if len(content) < 50:
            score -= 2

        if keywords:
            covered = sum(1 for kw in keywords
                         if kw.lower() in content.lower())
            score *= (covered / len(keywords))

        return min(max(score, 0), 10)
